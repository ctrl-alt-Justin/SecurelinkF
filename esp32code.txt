// Robust JSON field extraction utility that preserves whitespace and handles escape sequences
String extractJsonField(const String& json, const String& key) {
    int kIdx = json.indexOf("\"" + key + "\"");
    if (kIdx < 0) return "";
    int colIdx = json.indexOf(":", kIdx + key.length() + 2);
    if (colIdx < 0) return "";
    
    // Find first non-whitespace character after colon
    int valStart = colIdx + 1;
    while (valStart < (int)json.length() && (json[valStart] == ' ' || json[valStart] == '\t' || json[valStart] == '\r' || json[valStart] == '\n')) {
        valStart++;
    }
    if (valStart >= (int)json.length()) return "";

    if (json[valStart] == '\"') {
        // Quoted string value: preserve exact content between quotes, unescaping backslash sequences
        int cur = valStart + 1;
        String val = "";
        while (cur < (int)json.length()) {
            char c = json[cur];
            if (c == '\\' && cur + 1 < (int)json.length()) {
                char nextC = json[cur + 1];
                if (nextC == '\"') { val += '\"'; cur += 2; continue; }
                else if (nextC == '\\') { val += '\\'; cur += 2; continue; }
                else if (nextC == '/') { val += '/'; cur += 2; continue; }
                else if (nextC == 'b') { val += '\b'; cur += 2; continue; }
                else if (nextC == 'f') { val += '\f'; cur += 2; continue; }
                else if (nextC == 'n') { val += '\n'; cur += 2; continue; }
                else if (nextC == 'r') { val += '\r'; cur += 2; continue; }
                else if (nextC == 't') { val += '\t'; cur += 2; continue; }
            } else if (c == '\"') {
                // Closing quote found! Return exact string without trim!
                return val;
            }
            val += c;
            cur++;
        }
        return val;
    } else {
        // Unquoted value (number, boolean, null)
        int end = valStart;
        while (end < (int)json.length() && json[end] != ',' && json[end] != '}' && json[end] != ']' && json[end] != ' ' && json[end] != '\r' && json[end] != '\n') {
            end++;
        }
        return json.substring(valStart, end);
    }
}

// --- STRUCTURED SERVER ENDPOINT PARSER ---
struct ServerEndpoint {
    String raw;
    String scheme; // "http", "https", "ws", "wss"
    String host;
    int port;
    String path;
    bool isTls;
};

ServerEndpoint parseServerEndpoint(const String& rawUrl, int defaultPort = 8080) {
    ServerEndpoint ep;
    ep.raw = rawUrl;
    ep.port = defaultPort;
    ep.path = "/";
    ep.scheme = "http";
    ep.isTls = false;

    String clean = rawUrl;
    clean.trim();

    if (clean.startsWith("https://")) {
        ep.scheme = "https";
        ep.isTls = true;
        clean = clean.substring(8);
        ep.port = 443;
    } else if (clean.startsWith("http://")) {
        ep.scheme = "http";
        clean = clean.substring(7);
        ep.port = 8080;
    } else if (clean.startsWith("wss://")) {
        ep.scheme = "wss";
        ep.isTls = true;
        clean = clean.substring(6);
        ep.port = 443;
    } else if (clean.startsWith("ws://")) {
        ep.scheme = "ws";
        clean = clean.substring(5);
        ep.port = 8080;
    }

    int slashIdx = clean.indexOf('/');
    if (slashIdx >= 0) {
        ep.path = clean.substring(slashIdx);
        clean = clean.substring(0, slashIdx);
    }

    int colonIdx = clean.indexOf(':');
    if (colonIdx >= 0) {
        ep.host = clean.substring(0, colonIdx);
        int parsedPort = clean.substring(colonIdx + 1).toInt();
        if (parsedPort > 0) ep.port = parsedPort;
    } else {
        ep.host = clean;
    }

    if (ep.host.indexOf("trycloudflare.com") >= 0 || ep.port == 443) {
        ep.isTls = true;
        if (ep.port == 8080) ep.port = 443;
    }

    return ep;
}

// Bounded SNTP clock synchronization with zero unverified/insecure fallback
bool syncNtpTime(uint32_t timeoutMs = 12000) {
    Serial.println("[NTP] Synchronizing device clock with NTP pool (pool.ntp.org, time.google.com, time.cloudflare.com)...");
    configTime(0, 0, "pool.ntp.org", "time.google.com", "time.cloudflare.com");
    time_t now = time(nullptr);
    unsigned long start = millis();
    while (now < 1704067200 && (millis() - start < timeoutMs)) { // Jan 1, 2024 00:00:00 UTC
        delay(200);
        now = time(nullptr);
    }
    if (now >= 1704067200) {
        struct tm ti;
        gmtime_r(&now, &ti);
        Serial.printf("[NTP] Device clock verified: %04d-%02d-%02d %02d:%02d:%02d UTC (TLS validation active)\n",
            ti.tm_year + 1900, ti.tm_mon + 1, ti.tm_mday, ti.tm_hour, ti.tm_min, ti.tm_sec);
        return true;
    } else {
        Serial.println("[NTP] ERROR: Clock synchronization failed. TLS certificate verification cannot proceed.");
        return false;
    }
}

void connectWebSocket(const String& host, int port) {
    ServerEndpoint ep = parseServerEndpoint(host, port);
    String cleanHost = ep.host;
    int targetPort = ep.port;
    bool isSsl = ep.isTls;
    
    // Construct authenticated WebSocket query path (?device_id=PIN-XXXX&token=...)
    String wsPath = String(WS_PATH_PRIMARY) + "?device_id=" + pin_device_id;
    if (pin_device_token.length() > 0) {
        wsPath += "&token=" + pin_device_token;
        String extraHdrs = "X-Device-Id: " + pin_device_id + "\r\nX-Device-Token: " + pin_device_token + "\r\n";
        ws.setExtraHeaders(extraHdrs.c_str());
    } else {
        String extraHdrs = "X-Device-Id: " + pin_device_id + "\r\n";
        ws.setExtraHeaders(extraHdrs.c_str());
    }

    if (isSsl) {
        Serial.printf("[WS] Connecting via WSS/SSL to %s:%d%s [token=%s]\n", cleanHost.c_str(), targetPort, WS_PATH_PRIMARY, pin_device_token.length() > 0 ? "AUTHENTICATED" : "NONE");
        time_t now = time(nullptr);
        if (now < 1704067200) {
            // Attempt prompt clock sync before TLS WebSocket handshake
            syncNtpTime(4000);
        }
        ws.beginSslWithCA(cleanHost.c_str(), targetPort, wsPath.c_str(), cloudflare_ca_bundle);
    } else {
        Serial.printf("[WS] Connecting via WS to %s:%d%s [token=%s]\n", cleanHost.c_str(), targetPort, WS_PATH_PRIMARY, pin_device_token.length() > 0 ? "AUTHENTICATED" : "NONE");
        ws.begin(cleanHost.c_str(), targetPort, wsPath.c_str());
    }
    ws.setReconnectInterval(5000);
    ws.onEvent(onWebSocketEvent);
    hasPendingWsConnect = true;
    wsConnectStartedAt = millis();
}

WifiConnectResult connectToWifi(const String& targetSsid, const String& targetPass, unsigned long timeoutMs = 25000, int maxRetries = 2) {
    if (targetSsid.length() == 0) {
        return {false, "ERR_PROVISIONING_INVALID", "Wi-Fi SSID cannot be empty", ""};
    }

    Serial.printf("\n[WIFI-CONN] Attempting connection to '%s' (2.4 GHz, timeout: %lu ms, retries: %d)...\n", 
                  targetSsid.c_str(), timeoutMs, maxRetries);

    WiFi.disconnect(true);
    vTaskDelay(pdMS_TO_TICKS(150));
    WiFi.mode(WIFI_STA);

    unsigned long perAttemptTimeout = timeoutMs / maxRetries;
    if (perAttemptTimeout < 10000) perAttemptTimeout = 10000;

    for (int attempt = 1; attempt <= maxRetries; attempt++) {
        Serial.printf("[WIFI-CONN] Connection attempt %d/%d to '%s'...\n", attempt, maxRetries, targetSsid.c_str());
        lastWifiDisconnectReason = 0;
        wifiStaConnected = false;
        wifiGotIp = false;

        WiFi.begin(targetSsid.c_str(), targetPass.c_str());

        unsigned long tStart = millis();
        while (millis() - tStart < perAttemptTimeout) {
            vTaskDelay(pdMS_TO_TICKS(150));
            // Confirm BOTH status WL_CONNECTED and valid non-zero IP
            if (WiFi.status() == WL_CONNECTED && WiFi.localIP() != IPAddress(0, 0, 0, 0)) {
                WiFi.setSleep(false);
                String assignedIp = WiFi.localIP().toString();
                Serial.printf("[WIFI-CONN] Connection successful on attempt %d! IP: %s (RSSI: %d dBm)\n", 
                              attempt, assignedIp.c_str(), WiFi.RSSI());
                return {true, "", "", assignedIp};
            }

            // Early exit if access point explicitly rejects credentials
            if (lastWifiDisconnectReason == 202 || lastWifiDisconnectReason == 15 || 
                lastWifiDisconnectReason == 2 || lastWifiDisconnectReason == 14) {
                Serial.printf("[WIFI-CONN] Authentication rejected by AP (reason %d)!\n", lastWifiDisconnectReason);
                return {false, "ERR_AUTH_REJECTED", "Wi-Fi password rejected by access point '" + targetSsid + "'", ""};
            }

            // Early exit if AP does not exist / not found
            if (lastWifiDisconnectReason == 201) {
                Serial.printf("[WIFI-CONN] Access point '%s' not found (reason 201)!\n", targetSsid.c_str());
                if (attempt == maxRetries) {
                    return {false, "ERR_AP_NOT_FOUND", "Access point '" + targetSsid + "' not found. Ensure 2.4 GHz network is in range", ""};
                }
            }
        }

        if (attempt < maxRetries) {
            Serial.printf("[WIFI-CONN] Attempt %d failed. Retrying in 1.5s...\n", attempt);
            WiFi.disconnect(true);
            vTaskDelay(pdMS_TO_TICKS(1500));
        }
    }

    // Retries exhausted — diagnose specific failure mode
    if (wifiStaConnected && !wifiGotIp) {
        Serial.println("[WIFI-CONN] Associated with AP, but DHCP lease timed out!");
        return {false, "ERR_DHCP_TIMEOUT", "Associated with Wi-Fi, but failed to obtain IP address from DHCP", ""};
    }
    if (lastWifiDisconnectReason == 201) {
        return {false, "ERR_AP_NOT_FOUND", "Access point '" + targetSsid + "' was not found. Verify SSID and 2.4 GHz availability", ""};
    }
    if (lastWifiDisconnectReason == 202 || lastWifiDisconnectReason == 15 || 
        lastWifiDisconnectReason == 2 || lastWifiDisconnectReason == 14) {
        return {false, "ERR_AUTH_REJECTED", "Wi-Fi password rejected by access point", ""};
    }

    return {false, "ERR_WIFI_TIMEOUT", "Wi-Fi connection timed out while connecting to '" + targetSsid + "'", ""};
}

// 1. 2nd-Order Butterworth High-Pass Filter (100Hz cutoff at 16kHz)
// Strips sub-audible fan rumble & physical handling noise (<100Hz) while keeping 100% of vocal fundamentals (100Hz-3400Hz)
inline int16_t applyHighPass(int16_t input) {
    static float x1 = 0.0f, x2 = 0.0f, y1 = 0.0f, y2 = 0.0f;
    float x0 = (float)input;
    float y0 = 0.972614f * x0 - 1.945228f * x1 + 0.972614f * x2 + 1.944478f * y1 - 0.945978f * y2;
    x2 = x1; x1 = x0;
    y2 = y1; y1 = y0;
    if (y0 > 32767.0f) y0 = 32767.0f;
    else if (y0 < -32768.0f) y0 = -32768.0f;
    return (int16_t)y0;
}

// 2. Soft Limiter (Prevents square-wave distortion on loud speech for Whisper)
inline int16_t applySoftLimiter(int32_t sample) {
    const int32_t limit = 26000;
    if (sample > limit) {
        int32_t val = limit + ((sample - limit) >> 2);
        return (int16_t)(val > 32767 ? 32767 : val);
    } else if (sample < -limit) {
        int32_t val = -limit + ((sample + limit) >> 2);
        return (int16_t)(val < -32768 ? -32768 : val);
    }
    return (int16_t)sample;
}

// --- BATTERY TELEMETRY (XIAO ESP32S3 Sense) ---
// Note: Standard Seeed Studio XIAO ESP32-S3 boards do NOT route VBAT internally.
// An external 1:2 voltage divider (2x 100kΩ or 2x 200kΩ resistors) between BAT+ and GND,
// with midpoint wired to A0 (GPIO 1), is required for physical battery measurement.
#define BATTERY_PIN 1

static unsigned long totalStreamTimeMs = 0;
static unsigned long streamStartMs = 0;

struct LiPoCurvePoint {
    int mv;
    int pct;
};

// Calibrated non-linear discharge curve for standard 3.7V LiPo cell
static const LiPoCurvePoint LIPO_CALIBRATION_CURVE[] = {
    {4200, 100},
    {4150, 95},
    {4110, 90},
    {4080, 85},
    {4020, 80},
    {3980, 75},
    {3950, 70},
    {3910, 65},
    {3870, 60},
    {3850, 55},
    {3840, 50},
    {3820, 45},
    {3800, 40},
    {3790, 35},
    {3770, 30},
    {3750, 25},
    {3730, 20},
    {3710, 15},
    {3690, 10},
    {3610, 5},
    {3400, 1},
    {3270, 0}
};
static const size_t LIPO_CURVE_POINTS = sizeof(LIPO_CALIBRATION_CURVE) / sizeof(LIPO_CALIBRATION_CURVE[0]);

static int estimateLipoPercentageFromMv(int mv) {
    if (mv >= 4200) return 100;
    if (mv <= 3270) return 0;
    for (size_t i = 0; i < LIPO_CURVE_POINTS - 1; i++) {
        if (mv <= LIPO_CALIBRATION_CURVE[i].mv && mv >= LIPO_CALIBRATION_CURVE[i + 1].mv) {
            int vHigh = LIPO_CALIBRATION_CURVE[i].mv;
            int vLow = LIPO_CALIBRATION_CURVE[i + 1].mv;
            int pHigh = LIPO_CALIBRATION_CURVE[i].pct;
            int pLow = LIPO_CALIBRATION_CURVE[i + 1].pct;
            return pLow + ((mv - vLow) * (pHigh - pLow)) / (vHigh - vLow);
        }
    }
    return 0;
}

static float emaFilteredMv = -1.0f;

BatteryTelemetry readBatteryTelemetry() {
    analogReadResolution(12);
    analogSetPinAttenuation(BATTERY_PIN, ADC_11db);
    
    // Average 16 samples to filter high-frequency ADC noise
    uint32_t rawSum = 0;
    for (int i = 0; i < 16; i++) {
        rawSum += analogRead(BATTERY_PIN);
        delayMicroseconds(50);
    }
    uint32_t raw = rawSum / 16;
    
    // Factory-calibrated ESP32-S3 analogReadMilliVolts
    uint32_t pinMv = analogReadMilliVolts(BATTERY_PIN);
    
    // With 1:2 divider ratio (e.g. 2x 100kΩ or 2x 200kΩ resistors):
    int calculatedMv = (int)(pinMv * 2);
    
    // Valid LiPo operating voltage window under load/charging is 2800mV to 4400mV.
    // If voltage is < 2800mV, the pin is floating (no divider installed or disconnected).
    if (calculatedMv >= 2800 && calculatedMv <= 4400) {
        if (emaFilteredMv < 0) {
            emaFilteredMv = (float)calculatedMv;
        } else {
            // Smooth transient ADC noise with 80/20 EMA filter
            emaFilteredMv = (emaFilteredMv * 0.8f) + ((float)calculatedMv * 0.2f);
        }
        int filteredMv = (int)(emaFilteredMv + 0.5f);
        int pct = estimateLipoPercentageFromMv(filteredMv);
        return { (int)raw, filteredMv, pct };
    } else {
        // Floating or missing measurement circuit: preserve unknown state
        emaFilteredMv = -1.0f;
        return { (int)raw, -1, -1 };
    }
}

    return readBatteryTelemetry().batteryMv;

}

// Non-blocking snapshot engine:
// If camera hardware freezes or drops DMA, Core 0 audio streaming will NEVER hang!
static volatile bool snapshotInProgress = false;
static unsigned long snapshotStartedAt = 0;
static camera_fb_t* volatile readySnapshotFb = nullptr;

void snapshotWorkerTask(void* param) {
    // Drop the pre-buffered frame in FIFO to ensure the snapshot is fresh
    camera_fb_t * staleFb = esp_camera_fb_get();
    if (staleFb) {
        esp_camera_fb_return(staleFb);
    }

    // Capture fresh evidence snapshot from OV3660 sensor
    camera_fb_t * fb = esp_camera_fb_get();
    if (fb) {
        readySnapshotFb = fb;
        Serial.printf("[CAM] OV3660 snapshot captured (%d bytes, %dx%d)\n", fb->len, fb->width, fb->height);
    } else {
        Serial.println("[CAM] esp_camera_fb_get() returned null (capture failed).");
        snapshotInProgress = false;
    }
    vTaskDelete(NULL);
}

void captureIncidentSnapshot() {
    if (!cameraReady) {
        Serial.println("[CAM] Snapshot skipped: OV3660 camera not ready.");
        return;
    }
    if (snapshotInProgress) {
        if (millis() - snapshotStartedAt > 3500) {
            Serial.println("[CAM] Warning: Previous snapshot task timed out. Resetting state.");
            snapshotInProgress = false;
        } else {
            Serial.println("[CAM] Snapshot already in progress, skipping.");
            return;
        }
    }

    snapshotInProgress = true;
    snapshotStartedAt = millis();
    Serial.println("[CAM] Firing OV3660 snapshot task on Core 0...");
    BaseType_t res = xTaskCreatePinnedToCore(
        snapshotWorkerTask,
        "camSnapTask",
        4096,
        NULL,
        1,
        NULL,
        0
    );
    if (res != pdPASS) {
        snapshotInProgress = false;
        Serial.println("[CAM] Failed to create camSnapTask.");
    }
}

void onWebSocketEvent(WStype_t type, uint8_t* payload, size_t length) {
    switch (type) {
        case WStype_CONNECTED:
            websocketConnected = true;
            serverAuthenticated = false; // Transport connected, but authentication handshake pending!
            hasActiveStream = false; // Standby connection ready! Audio streaming begins only when voice is detected
            hasPendingWsConnect = false;
            isStreaming = false; // Always clear stale streaming state on connect
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            needImmediateHeartbeat = false; // Heartbeats sent only after authentication handshake succeeds
            Serial.println("[WS] Connected to Central System. Sending authentication handshake...");
            if (pin_device_token.length() > 0) {
                String authPayload = "{\"type\":\"auth\",\"device_id\":\"" + pin_device_id + "\",\"token\":\"" + pin_device_token + "\"}";
                ws.sendTXT(authPayload);
                Serial.printf("[WS] Transmitted auth frame for %s\n", pin_device_id.c_str());
            }
            break;
        case WStype_TEXT:
            if (length > 0) {
                String msg = String((char*)payload);
                if (msg == "TRIGGER:DANGER" || msg == "CMD:CAPTURE_SNAPSHOT") {
                    Serial.println("[CAM] Speech evaluated as DANGER! Capturing camera snapshot...");
                    captureIncidentSnapshot();
                } else if (msg.startsWith("CONFIG:BUFFER:")) {
                    int sec = msg.substring(14).toInt();
                    if (sec > 0) {
                        SILENCE_TIMEOUT = sec * 1000;
                        Serial.printf("[CFG] Silence Timeout updated to: %lu ms\n", SILENCE_TIMEOUT);
                    }
                } else if (msg.startsWith("CONFIG:WIFI:")) {
                    // Format: CONFIG:WIFI:<ssid>:<pass>:<server_ip>:<port>
                    int p1 = msg.indexOf(':', 12);
                    int p2 = msg.indexOf(':', p1 + 1);
                    int p3 = msg.indexOf(':', p2 + 1);
                    if (p1 > 0 && p2 > 0) {
                        String n_ssid = msg.substring(12, p1);
                        String n_pass = msg.substring(p1 + 1, p2);
                        String n_server = p3 > 0 ? msg.substring(p2 + 1, p3) : msg.substring(p2 + 1);
                        int n_port = p3 > 0 ? msg.substring(p3 + 1).toInt() : 8080;
                        if (n_port <= 0) n_port = 8080;

                        Serial.printf("\n[CFG] >>> RECEIVED NEW WI-FI CHECKPOINT OVER WEBSOCKET! <<<\n");
                        Serial.printf("[CFG] New SSID: '%s' | Server: '%s:%d'\n", n_ssid.c_str(), n_server.c_str(), n_port);

                        // Save current network as backup/secondary checkpoint before switching
                        if (wifi_ssid.length() > 0 && wifi_ssid != n_ssid) {
                            prefs.putString("ssid2", wifi_ssid);
                            prefs.putString("pass2", wifi_pass);
                            wifi_ssid2 = wifi_ssid;
                            wifi_pass2 = wifi_pass;
                        }

                        prefs.putString("ssid", n_ssid);
                        prefs.putString("pass", n_pass);
                        prefs.putString("server", n_server);
                        prefs.putInt("port", n_port);

                        wifi_ssid = n_ssid;
                        wifi_pass = n_pass;
                        server_host = n_server;
                        server_port = n_port;
                        cachedServerIP = n_server;

                        ws.sendTXT("OK:CONFIG_WIFI_SAVED");
                        delay(400);
                        ws.disconnect();
                        delay(200);
                        Serial.println("[SYS] Rebooting to connect to updated checkpoint...");
                        ESP.restart();
                    }
                } else if (msg == "CMD:FORGET_WIFI" || msg == "CMD:RESET_WIFI") {
                    Serial.println("\n[SYS] >>> RECEIVED 'CMD:FORGET_WIFI' FROM DASHBOARD! <<<");
                    Serial.println("[SYS] Erasing Wi-Fi credentials from NVS flash...");
                    prefs.remove("ssid");
                    prefs.remove("pass");
                    prefs.remove("ssid2");
                    prefs.remove("pass2");
                    prefs.remove("token");
                    wifi_ssid = "";
                    wifi_pass = "";
                    wifi_ssid2 = "";
                    wifi_pass2 = "";
                    pin_device_token = "";
                    hasStoredCredentials = false;
                    serverAuthenticated = false;
                    isDevicePaired = false;
                    ws.disconnect();
                    delay(500);
                    Serial.println("[SYS] Rebooting into Setup Portal...");
                    ESP.restart();
                } else if (msg.indexOf("\"type\":\"auth_ok\"") >= 0) {
                    String authDevId = extractJsonField(msg, "device_id");
                    if (authDevId.length() == 0 || authDevId == pin_device_id) {
                        serverAuthenticated = true;
                        isDevicePaired = true;
                        hasStoredCredentials = true;
                        needImmediateHeartbeat = true; // Safe to send heartbeats now
                        Serial.printf("[WS-AUTH] WebSocket Authentication Confirmed by Central System for %s!\n", pin_device_id.c_str());
                        deferBleText("{\"type\":\"pairing_success\",\"device_id\":\"" + pin_device_id + "\"}");
                    } else {
                        Serial.printf("[WS-AUTH] Rejected auth_ok for mismatched device_id: %s (expected %s)\n", authDevId.c_str(), pin_device_id.c_str());
                    }
                } else if (msg.indexOf("\"type\":\"auth_challenge\"") >= 0) {
                    if (pin_device_token.length() > 0) {
                        String authPayload = "{\"type\":\"auth\",\"device_id\":\"" + pin_device_id + "\",\"token\":\"" + pin_device_token + "\"}";
                        ws.sendTXT(authPayload);
                        Serial.println("[WS-AUTH] Responded to auth challenge with token frame.");
                    }
                } else if (msg.indexOf("\"type\":\"auth_error\"") >= 0 || msg.indexOf("\"ERR_AUTH_FAILED\"") >= 0) {
                    Serial.println("[WS-AUTH] WebSocket Authentication Rejected by Central System! Invalid or revoked token.");
                    serverAuthenticated = false;
                    hasStoredCredentials = false;
                    isDevicePaired = false;
                    prefs.remove("token");
                    pin_device_token = "";
                    ws.disconnect();
                    deferBleText("{\"type\":\"provision_error\",\"code\":\"ERR_WS_AUTH\",\"message\":\"Central System rejected device token\"}");
                }
            }
            break;
        case WStype_DISCONNECTED:
            websocketConnected = false;
            serverAuthenticated = false;
            hasActiveStream = false;
            hasPendingWsConnect = false;
            isStreaming = false; // Always clear streaming state on disconnect
            isUploadingArchive = false;
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            if (uploadFile) uploadFile.close();
            Serial.println("[WS] Disconnected");
            break;
        default:
            break;
    }
}

void setupMicrophone() {
    I2S.setPinsPdmRx(42, 41);
    if (!I2S.begin(I2S_MODE_PDM_RX, SAMPLE_RATE, I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO)) {
        Serial.println("[ERR] I2S PDM Microphone Init Failed!");
        micInitialized = false;
    } else {
        micInitialized = true;
        Serial.println("[AUDIO] MSM261D PDM Microphone initialized (16kHz 16-bit Mono).");
    }
    speex_st = speex_preprocess_state_init(FRAME_SIZE, SAMPLE_RATE);
    if (!speex_st) {
        Serial.println("[ERR] Speex VAD state init failed!");
        vadInitialized = false;
    } else {
        vadInitialized = true;
        int vadEnabled = 1;
        int probStart = 75;    // Balanced onset trigger
        int probContinue = 55; // Natural sentence decay
        speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_VAD, &vadEnabled);
        speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_START, &probStart);
        speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_CONTINUE, &probContinue);

        // Denoising is handled by Microsoft NSNet2 on the GPU/server.
        // Keeping Speex denoise OFF on the ESP32 frees 80% CPU on Core 1 and eliminates I2S DMA dropouts!
        int denoise = 0;
        speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_DENOISE, &denoise);
        Serial.println("[AUDIO] Speex VAD Engine initialized on Core 1.");
    }
}

void discoverCheckpoint() {
    if (hasActiveStream || hasPendingWsConnect || ws.isConnected() || wifiConnectInProgress) return;

    if (!hasStoredCredentials || pin_device_token.length() == 0) {
        static unsigned long lastUnpairedLog = 0;
        if (millis() - lastUnpairedLog > 10000) {
            lastUnpairedLog = millis();
            Serial.printf("[WS] Pin '%s' connected to Wi-Fi but unpaired (no token). Awaiting pairing via Mobile BLE or Setup Portal.\n", pin_device_id.c_str());
        }
        return;
    }

    if (WiFi.status() != WL_CONNECTED) {
        wifiConnectInProgress = true;
        wifiConnectStartedAt = millis();
        WiFi.mode(WIFI_STA);
        WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        Serial.printf("[WIFI] Connecting to %s...\n", wifi_ssid.c_str());
        return;
    }

    if (cachedServerIP != "") {
        connectWebSocket(cachedServerIP, server_port);
        return;
    }

    udp.begin(8888);
    udp.beginPacket(IPAddress(255, 255, 255, 255), 8888);
    udp.print("SECURELINK_PIN_V1");
    udp.endPacket();

    // Give server up to 800ms to respond over Wi-Fi (50ms was too fast)
    unsigned long waitStart = millis();
    while (millis() - waitStart < 800) {
        if (udp.parsePacket()) {
            cachedServerIP = udp.remoteIP().toString();
            Serial.printf("[UDP] Found Checkpoint Server at: %s\n", cachedServerIP.c_str());
            connectWebSocket(cachedServerIP, server_port);
            break;
        }
        delay(20);
    }
}

// Ensure space exists before local recording: deletes oldest timestamped incident
void enforceFreeSpace() {
    if (!fsReady) return;
    
    while ((LittleFS.totalBytes() - LittleFS.usedBytes()) < 400000) {
        File root = LittleFS.open("/");
        File file = root.openNextFile();
        if (!file) break;
        
        String oldestFile = "";
        unsigned long oldestTime = 0xFFFFFFFF;

        while (file) {
            String name = String(file.name());
            if (name.startsWith("inc_") && name.endsWith(".pcm")) {
                int start = name.indexOf('_') + 1;
                int end = name.indexOf('.');
                unsigned long t = name.substring(start, end).toInt();
                if (t < oldestTime) {
                    oldestTime = t;
                    oldestFile = "/" + name;
                }
            }
            file = root.openNextFile();
        }
        root.close();

        if (oldestFile != "") {
            LittleFS.remove(oldestFile);
            Serial.println("[FS] Low space. Removed oldest file: " + oldestFile);
        } else {
            break;
        }
    }
}

void processArchiveUpload() {
    // Only upload offline recordings when connected to WebSocket, idle (not actively streaming live speech), and flash is ready
    if (!ws.isConnected() || isStreaming || isOfflineRecording || !fsReady) return;

    if (!isUploadingArchive) {
        File root = LittleFS.open("/");
        File file = root.openNextFile();
        while (file) {
            String name = String(file.name());
            if (name.endsWith(".pcm")) {
                uploadFilePath = "/" + name;
                uploadFile = LittleFS.open(uploadFilePath, FILE_READ);
                isUploadingArchive = true;
                ws.sendTXT("ARCHIVE_START:" + uploadFilePath);
                Serial.printf("[INC] Syncing stored incident: %s\n", uploadFilePath.c_str());
                file.close();
                root.close();
                return;
            }
            file = root.openNextFile();
        }
        root.close();
        return; 
    }

    static unsigned long lastArchiveChunkMs = 0;
    if (millis() - lastArchiveChunkMs < 40) return; // 40ms timer pace = 51.2 KB/s (safely below server 80 KB/s rate limit)

    if (uploadFile.available()) {
        uint8_t buf[2048];
        size_t bytesRead = uploadFile.read(buf, sizeof(buf));
        if (bytesRead > 0) {
            ws.sendBIN(buf, bytesRead);
            lastArchiveChunkMs = millis();
        }
    } else {
        uploadFile.close();
        LittleFS.remove(uploadFilePath);
        ws.sendTXT("ARCHIVE_END:" + uploadFilePath);
        Serial.printf("[INC] Sync complete. Deleted: %s\n", uploadFilePath.c_str());
        isUploadingArchive = false;
        lastArchiveChunkMs = 0;
    }
}

void handleSerialCommands() {
    while (Serial.available()) {
        String line = Serial.readStringUntil('\n');
        line.trim();
        if (line.length() == 0) continue;

        if (line == "SL:HANDSHAKE:REQ" || line.startsWith("SL:HANDSHAKE")) {
            Serial.printf("SL:IDENT:%s:%s:2.1\n", pin_device_id.c_str(), pin_mac_address.c_str());
            Serial.flush();
        } else if (line == "SL:AUTH:OK") {
            usbAuthenticated = true;
            lastUsbHeartbeatMs = millis();
            Serial.println("[USB] Authenticated with SecureLink Central System!");
        } else if (line.startsWith("SL:AUTH:ERR") || line.startsWith("SL:AUTH:FAIL")) {
            usbAuthenticated = false;
            Serial.println("[USB] Authorization failed!");
        } else if (line.startsWith("SL:AUDIO:ACK:")) {
            lastAudioAckSeq = (uint32_t)line.substring(13).toInt();
        } else if (line == "SL:PONG") {
            lastUsbPongMs = millis();
        } else if (line == "SL:VAD:DISARM" || line.startsWith("SL:VAD:DISARM:")) {
            vadExplicitlyDisarmed = true;
            speechArmed = false;
            int colonIdx = line.indexOf(':', 14);
            vadDisarmReason = colonIdx > 0 ? line.substring(colonIdx + 1) : (line.length() > 14 ? line.substring(14) : "User requested disarm");
            Serial.printf("[VAD] VAD Engine DISARMED: %s\n", vadDisarmReason.c_str());
            Serial.printf("SL:VAD:DISARMED:%s\n", vadDisarmReason.c_str());
            Serial.flush();
        } else if (line == "SL:VAD:ARM") {
            if (micInitialized && vadInitialized) {
                vadExplicitlyDisarmed = false;
                vadDisarmReason = "OPERATIONAL";
                Serial.println("[VAD] VAD Engine ARMED.");
                Serial.println("SL:VAD:ARMED:OK");
            } else {
                Serial.printf("[VAD] Cannot arm: micInit=%d, vadInit=%d\n", micInitialized ? 1 : 0, vadInitialized ? 1 : 0);
                Serial.printf("SL:VAD:ARM:FAIL:mic=%d:vad=%d\n", micInitialized ? 1 : 0, vadInitialized ? 1 : 0);
            }
            Serial.flush();
        } else if (line == "SL:VAD:RECOVER" || line == "SL:AUDIO:RECOVER") {
            Serial.println("[VAD] Performing audio & VAD pipeline recovery...");
            if (!micInitialized) {
                I2S.end();
                delay(50);
                I2S.setPinsPdmRx(42, 41);
                micInitialized = I2S.begin(I2S_MODE_PDM_RX, SAMPLE_RATE, I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO);
            }
            if (!vadInitialized && speex_st == nullptr) {
                speex_st = speex_preprocess_state_init(FRAME_SIZE, SAMPLE_RATE);
                if (speex_st) {
                    vadInitialized = true;
                    int vadEnabled = 1;
                    int probStart = 75;
                    int probContinue = 55;
                    int denoise = 0;
                    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_VAD, &vadEnabled);
                    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_START, &probStart);
                    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_CONTINUE, &probContinue);
                    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_DENOISE, &denoise);
                }
            }
            if (micInitialized && vadInitialized) {
                vadExplicitlyDisarmed = false;
                vadDisarmReason = "OPERATIONAL";
                Serial.println("[VAD] Recovery successful: Audio pipeline active & armed.");
                Serial.println("SL:VAD:RECOVER:OK");
            } else {
                Serial.printf("[VAD] Recovery incomplete: mic=%d vad=%d\n", micInitialized ? 1 : 0, vadInitialized ? 1 : 0);
                Serial.printf("SL:VAD:RECOVER:FAIL:mic=%d:vad=%d\n", micInitialized ? 1 : 0, vadInitialized ? 1 : 0);
            }
            Serial.flush();
        } else if (line == "SL:TRIGGER:DANGER") {
            Serial.println("[USB] *** THREAT ALERT: TRIGGER DANGER DETECTED ***");
        } else if (line.startsWith("{") && line.indexOf("\"provision\"") >= 0) {
            Serial.println("{\"type\":\"ack\",\"cmd\":\"provision\",\"status\":\"received\"}");
            String n_ssid = extractJsonField(line, "ssid");
            String n_pass = extractJsonField(line, "password");
            String n_server = extractJsonField(line, "server");
            String n_pcode = extractJsonField(line, "pairing_code");
            String s_port = extractJsonField(line, "port");
            int n_port = s_port.length() > 0 ? s_port.toInt() : 8080;
            if (n_port <= 0) n_port = 8080;
            if (n_ssid.length() > 0) {
                prefs.putString("ssid", n_ssid);
                prefs.putString("pass", n_pass);
                if (n_server.length() > 0) prefs.putString("server", n_server);
                prefs.putInt("port", n_port);
                if (n_pcode.length() >= 6) prefs.putString("pcode_pending", n_pcode);
                wifi_ssid = n_ssid;
                wifi_pass = n_pass;
                server_host = n_server;
                server_port = n_port;
                Serial.println("{\"type\":\"ack\",\"cmd\":\"provision\",\"status\":\"success\"}");
            }
        }
    }
}

// =========================================================================
// CORE 1: DEDICATED REAL-TIME AUDIO ENGINE
// Continuously samples I2S, executes DSP + Speex VAD, and appends to PSRAM
// ring buffer. Completely isolated from Wi-Fi/TCP jitter on Core 0!
// =========================================================================
void audioRecordTask(void* pvParameters) {
    Serial.println("[AUDIO] Core 1 Audio Engine task running at Priority 5.");
    while (true) {
        size_t bytesRead = I2S.readBytes((char*)audioBuffer, sizeof(audioBuffer));
        if (bytesRead != sizeof(audioBuffer)) {
            vTaskDelay(pdMS_TO_TICKS(2));
            continue;
        }
        audioFramesCaptured++;

        // 1. Apply High-Pass Filter & Peak Tracking
        int32_t currentPeak = 0;
        for (int i = 0; i < FRAME_SIZE; i++) {
            int16_t filtered = applyHighPass(audioBuffer[i]);
            audioBuffer[i] = filtered; // Clean audio for transmission
            vadBuffer[i] = filtered;
            if (abs(filtered) > currentPeak) currentPeak = abs(filtered);
        }
        lastFramePeak = currentPeak;

        // 2. Speex VAD
        int isSpeech = 0;
        if (speex_st != nullptr) {
            speex_preprocess_run(speex_st, vadBuffer);
            speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_GET_VAD, &isSpeech);
        }

        // Speech detection criteria:
        // 1) Speex VAD confirms vocal fold harmonics (isSpeech != 0) with peak above ambient baseline (+35) and >= 135
        // 2) Amplitude fallback bypass (currentPeak >= 380): detects loud voice or exclamation without Speex delay
        bool speechDetected = ((isSpeech != 0) && (currentPeak > ambientFloor + 35) && (currentPeak >= 135)) || (currentPeak >= 380);

        // VAD Hangover Engine: 400ms (20 frames) hangover bridges natural breath pauses & consonants
        if (speechDetected) {
            if (speechStreak < 4) speechStreak++;
            if (speechStreak >= SPEECH_STREAK_REQUIRED) {
                hangoverFrames = 20;
            }
        } else {
            if (speechStreak > 0) speechStreak--;
            if (hangoverFrames > 0) hangoverFrames--;
        }

        speechArmed = !vadExplicitlyDisarmed && ((speechStreak >= SPEECH_STREAK_REQUIRED) || (hangoverFrames > 0));

        // Dynamic ambient floor tracking: adaptively track background when speech is inactive
        if (!speechArmed && isSpeech == 0) {
            ambientFloor = (ambientFloor * 31 + currentPeak) / 32;
            if (ambientFloor < 20) ambientFloor = 20;
            if (ambientFloor > 160) ambientFloor = 160;
        }

        if (speechArmed != lastVadState) {
            Serial.println(speechArmed ? "[VAD] VOICE DETECTED" : "[VAD] SILENCE");
            lastVadState = speechArmed;
        }

        // 3. Format clean 16-bit PCM for transmission
        // Passing full natural dynamic range directly preserves vocal warmth and eliminates raspy distortion
        uint8_t outBuf[FRAME_SIZE * sizeof(int16_t)];
        for (int i = 0; i < FRAME_SIZE; i++) {
            int16_t sample = audioBuffer[i];
            outBuf[i * 2] = sample & 0xFF;
            outBuf[i * 2 + 1] = (sample >> 8) & 0xFF;
        }

        // 4. Lock-free append to PSRAM Ring Buffer
        if (audioRingBuffer != nullptr) {
            for (size_t i = 0; i < sizeof(outBuf); i++) {
                audioRingBuffer[(ringHead + i) % RING_BUFFER_SIZE] = outBuf[i];
            }
            ringHead = (ringHead + sizeof(outBuf)) % RING_BUFFER_SIZE;

            // When NOT streaming, keep ringTail floating exactly PRE_ROLL_BYTES behind ringHead
            if (!isStreaming) {
                ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            } else {
                // Buffer Overrun Guard: If ringHead gets dangerously close to ringTail (e.g. Wi-Fi stalled),
                // advance ringTail to drop oldest unread data cleanly and protect live speech continuity.
                size_t used = (ringHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;
                if (used >= RING_BUFFER_SIZE - (BATCH_SIZE * 2)) {
                    ringTail = (ringTail + BATCH_SIZE) % RING_BUFFER_SIZE;
                }
            }
        }

        taskYIELD(); // Yield FreeRTOS slice to avoid starving RTOS scheduler and watchdogs
    }
}

void setup() {
    // 1. BALANCED 160MHz CLOCK FOR OPTIMAL BATTERY LIFE & LOW THERMAL FOOTPRINT
    setCpuFrequencyMhz(160);

    Serial.begin(115200);
    Serial.setTimeout(20);
    delay(500);
    Serial.printf("[SYS] CPU Clock: %d MHz\n", getCpuFrequencyMhz());

    pinMode(0, INPUT_PULLUP); // XIAO ESP32S3 Boot button
    loadPreferences();
    initDeviceIdentity(); // Compute persistent hardware ID & BLE device name

    // 2. CRITICAL FIX: INITIALIZE BLE FIRST!
    // BLE is active immediately upon boot so phone can discover and pair without buttons or cables
    setupBLE();

    // If Boot button is manually held at power-on, enter fallback captive portal
    if (digitalRead(0) == LOW) {
        Serial.println("[SYS] Boot button held at power-on! Starting manual setup portal...");
        startConfigPortal();
    }

    bootStartMs = millis();

    // 3. INITIALIZE PSRAM RING BUFFER
    if (psramInit()) {
        audioRingBuffer = (uint8_t*)ps_malloc(RING_BUFFER_SIZE);
        Serial.printf("[SYS] 8MB PSRAM found! Ring Buffer allocated in PSRAM: %u bytes\n", (unsigned int)RING_BUFFER_SIZE);
    } else {
        audioRingBuffer = (uint8_t*)malloc(32000);
        Serial.printf("[WARN] PSRAM not found. Allocated 32000 bytes in SRAM\n");
    }

    if (!audioRingBuffer) {
        Serial.println("[ERR] Ring buffer allocation failed!");
        while(1) delay(100);
    }

    setupMicrophone();
    setupCamera();

    // Launch Dedicated Audio Engine on Core 1 (Core 0 handles Wi-Fi, WebSocket, LittleFS, and System)
    xTaskCreatePinnedToCore(
        audioRecordTask,
        "AudioEngine",
        8192,
        NULL,
        5,
        &audioTaskHandle,
        1
    );
    
    if (LittleFS.begin(true)) {
        fsReady = true;
        Serial.printf("[FS] Flash mounted. Total: %u bytes, Used: %u bytes\n", 
            LittleFS.totalBytes(), LittleFS.usedBytes());
    } else {
        Serial.println("[ERR] LittleFS Mount Failed");
    }

    // 4. USB-ONLY OPERATION MODE (WI-FI & WEBSOCKET NETWORKING BYPASS)
    Serial.println("[USB] Direct USB-Only Mode Active. Awaiting Central System Handshake...");
}

void loop() {
    handleSerialCommands();

    // --- USB CDC CONNECTION DETECTION & AUTO-HANDSHAKE ---
    if (Serial) {
        if (!usbConnected) {
            usbConnected = true;
            Serial.printf("[USB] Port active. Identifying device as %s...\n", pin_device_id.c_str());
            Serial.printf("SL:IDENT:%s:%s:2.1\n", pin_device_id.c_str(), pin_mac_address.c_str());
            Serial.flush();
        }
    } else {
        if (usbConnected) {
            usbConnected = false;
            usbAuthenticated = false;
            if (isStreaming) {
                isStreaming = false;
            }
        }
    }

    // --- USB PERIODIC HEARTBEAT (every 2.0s) ---
    if (usbAuthenticated) {
        if (millis() - lastUsbHeartbeatMs >= 2000) {
            lastUsbHeartbeatMs = millis();
            BatteryTelemetry bt = readBatteryTelemetry();
            const char* vadStr = vadExplicitlyDisarmed ? "VAD DISARMED" : (speechArmed ? "SPEECH DETECTED" : (vadInitialized ? "VAD ARMED" : "VAD INIT FAIL"));
            Serial.printf("SL:PING:%d:%d:%s:%d:%d:%lu:%d:%s\n",
                bt.batteryPct,
                bt.batteryMv,
                vadStr,
                micInitialized ? 1 : 0,
                vadInitialized ? 1 : 0,
                (unsigned long)audioFramesCaptured,
                (int)lastFramePeak,
                vadDisarmReason.c_str()
            );
            Serial.flush();
        }
    }

    // Flush any deferred BLE notifications queued from inside onWrite callbacks.
    // This MUST run from loop() context, never from inside a BLE callback.
    flushDeferredBleTx();

    // Wi-Fi / WebSocket trigger bypassed in USB-only mode

    // Check for Boot button held for 2 seconds anytime to enter setup portal
    static unsigned long btnHoldStart = 0;
    if (digitalRead(0) == LOW) {
        if (btnHoldStart == 0) btnHoldStart = millis();
        else if (millis() - btnHoldStart > 2000) {
            btnHoldStart = 0;
            Serial.println("\n[SYS] Boot button held for 2s! Launching Setup Portal...");
            startConfigPortal();
        }
    } else {
        btnHoldStart = 0;
    }

    // Maintain BLE advertising if phone companion disconnected
    if (!bleClientConnected && oldBleClientConnected) {
        delay(300);
        pBleServer->startAdvertising();
        Serial.println("[BLE] Restarted advertising after companion disconnect.");
        oldBleClientConnected = bleClientConnected;
    }
    if (bleClientConnected && !oldBleClientConnected) {
        oldBleClientConnected = bleClientConnected;
    }

    // Check for completed non-blocking snapshot frame (Wi-Fi & BLE Multi-Channel Routing)
    if (readySnapshotFb != nullptr) {
        camera_fb_t* fb = readySnapshotFb;
        readySnapshotFb = nullptr;
        if (ws.isConnected() && serverAuthenticated) {
            // Wi-Fi Route: WebSocket Text Header + Binary Buffer
            String header = "SNAP:" + String(fb->len);
            ws.sendTXT(header);
            ws.sendBIN(fb->buf, fb->len);
            Serial.printf("[CAM] Incident snapshot sent to server over Wi-Fi (%d bytes)\n", fb->len);
        } else if (bleClientConnected) {
            // BLE Route: Nordic UART Service sliced to MTU 512
            String header = "SNAP:" + String(fb->len);
            sendBleText(header);
            delay(20);
            sendBleChunk(fb->buf, fb->len);
            Serial.printf("[CAM] Incident snapshot sent to Mobile Companion over BLE (%d bytes)\n", fb->len);
        }
        esp_camera_fb_return(fb);
        snapshotInProgress = false;
    }

    // Wi-Fi maintenance and WebSocket loop bypassed in USB-only mode

    // --- AUDIO STREAMING & RECORDING PIPELINE (USB PRIORITY WITH OFFLINE FALLBACK) ---
    // TRIGGER STREAMING WHEN VOICE IS DETECTED
    if (speechArmed) {
        lastSpeechTime = millis();
        if (!isStreaming) {
            if (usbAuthenticated) {
                isStreaming = true;
                usbStreamSeq++;
                usbChunkSeq = 0;
                totalStreamBytes = 0;
                streamStartMs = millis();
                Serial.printf("SL:STREAM_START:%u\n", (unsigned int)usbStreamSeq);
                Serial.flush();
                Serial.printf("[VAD] Voice triggered! Streaming via USB (Stream #%u).\n", (unsigned int)usbStreamSeq);
            } else if (ws.isConnected() && serverAuthenticated) {
                hasActiveStream = true;
                isStreaming = true;
                WiFi.setSleep(false);
                streamStartMs = millis();
                Serial.println("[VAD] Voice triggered! Streaming via Wi-Fi WebSocket.");
            } else if (!wifiConnectInProgress && fsReady) {
                enforceFreeSpace();
                isOfflineRecording = true;
                isStreaming = true;
                streamStartMs = millis();
                currentIncidentPath = "/inc_" + String(millis()) + ".pcm";
                offlineIncidentFile = LittleFS.open(currentIncidentPath, FILE_WRITE);
                Serial.printf("[VAD] Recording locally to %s.\n", currentIncidentPath.c_str());
            }
        }
    }

    // ACTIVE STREAMING / LOCAL RECORDING ENGINE (USB FRAMED PCM TRANSMISSION)
    if (isStreaming) {
        size_t available = (ringHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;

        bool maxStreamReached = (streamStartMs > 0 && (millis() - streamStartMs > 8000));
        if ((millis() - lastSpeechTime > SILENCE_TIMEOUT) || maxStreamReached) {
            // Voice ended or max stream limit reached: drain remaining audio with verified delivery
            size_t targetHead = ringHead;
            size_t toDrain = (targetHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;
            int failedAttempts = 0;
            while (toDrain >= sizeof(int16_t) && failedAttempts < 5) {
                if (!usbAuthenticated && !ws.isConnected() && !isOfflineRecording) break;
                size_t chunk = min((size_t)BATCH_SIZE, toDrain);
                uint8_t tempBuf[chunk];
                for (size_t i = 0; i < chunk; i++) {
                    tempBuf[i] = audioRingBuffer[(ringTail + i) % RING_BUFFER_SIZE];
                }

                bool sendOk = false;
                if (usbAuthenticated) {
                    usbChunkSeq++;
                    uint32_t currentSeq = usbChunkSeq;
                    Serial.printf("SL:AUDIO:%u:%u\n", (unsigned int)chunk, (unsigned int)currentSeq);
                    Serial.write(tempBuf, chunk);
                    Serial.flush();
                    unsigned long ackWait = millis();
                    while (millis() - ackWait < 150) {
                        if (Serial.available()) {
                            handleSerialCommands();
                        }
                        if (lastAudioAckSeq == currentSeq) {
                            sendOk = true;
                            break;
                        }
                        delay(2);
                    }
                    if (!sendOk) {
                        Serial.printf("SL:AUDIO:%u:%u\n", (unsigned int)chunk, (unsigned int)currentSeq);
                        Serial.write(tempBuf, chunk);
                        Serial.flush();
                        ackWait = millis();
                        while (millis() - ackWait < 150) {
                            if (Serial.available()) {
                                handleSerialCommands();
                            }
                            if (lastAudioAckSeq == currentSeq) {
                                sendOk = true;
                                break;
                            }
                            delay(2);
                        }
                    }
                    if (sendOk) {
                        totalStreamBytes += chunk;
                    } else if (fsReady) {
                        if (!isOfflineRecording) {
                            enforceFreeSpace();
                            isOfflineRecording = true;
                            char filename[40];
                            snprintf(filename, sizeof(filename), "/offline_inc_%lu.pcm", (unsigned long)millis());
                            offlineIncidentFile = LittleFS.open(filename, FILE_WRITE);
                        }
                        if (offlineIncidentFile) {
                            offlineIncidentFile.write(tempBuf, chunk);
                        }
                    }
                } else if (hasActiveStream && ws.isConnected() && serverAuthenticated) {
                    sendOk = ws.sendBIN(tempBuf, chunk);
                } else if (isOfflineRecording && offlineIncidentFile) {
                    sendOk = (offlineIncidentFile.write(tempBuf, chunk) == chunk);
                }

                if (sendOk) {
                    ringTail = (ringTail + chunk) % RING_BUFFER_SIZE;
                    toDrain -= chunk;
                    failedAttempts = 0;
                    delay(2);
                } else {
                    failedAttempts++;
                    delay(5);
                }
            }

            if (usbAuthenticated) {
                Serial.printf("SL:STREAM_END:%u:%u\n", (unsigned int)usbStreamSeq, (unsigned int)totalStreamBytes);
                Serial.flush();
                Serial.printf("[USB] Stream #%u completed (%u bytes).\n", (unsigned int)usbStreamSeq, (unsigned int)totalStreamBytes);
            }
            if (hasActiveStream && !isUploadingArchive) {
                ws.sendTXT("STREAM_END");
                hasActiveStream = false;
            }
            if (isOfflineRecording) {
                isOfflineRecording = false;
                if (offlineIncidentFile) {
                    offlineIncidentFile.close();
                    Serial.println("[FS] Offline recording saved and closed.");
                }
            }
            if (streamStartMs > 0) {
                totalStreamTimeMs += (millis() - streamStartMs);
                streamStartMs = 0;
            }
            isStreaming = false;
            WiFi.setSleep(false);
            // Float tail back to pre-roll window
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
        } else {
            // Actively draining audio from ringTail to ringHead in steady, verified BATCH_SIZE chunks
            if (available >= BATCH_SIZE) {
                static unsigned long lastBatchSentAt = 0;
                unsigned long paceMs = (available > 16000) ? 5 : 20;
                if (millis() - lastBatchSentAt >= paceMs) {
                    uint8_t txBuf[BATCH_SIZE];
                    for (size_t i = 0; i < BATCH_SIZE; i++) {
                        txBuf[i] = audioRingBuffer[(ringTail + i) % RING_BUFFER_SIZE];
                    }

                    bool sendOk = false;
                    if (usbAuthenticated) {
                        usbChunkSeq++;
                        uint32_t currentSeq = usbChunkSeq;
                        Serial.printf("SL:AUDIO:%u:%u\n", (unsigned int)BATCH_SIZE, (unsigned int)currentSeq);
                        Serial.write(txBuf, BATCH_SIZE);
                        Serial.flush();
                        unsigned long ackWait = millis();
                        while (millis() - ackWait < 150) {
                            if (Serial.available()) {
                                handleSerialCommands();
                            }
                            if (lastAudioAckSeq == currentSeq) {
                                sendOk = true;
                                break;
                            }
                            delay(2);
                        }
                        if (!sendOk) {
                            Serial.printf("SL:AUDIO:%u:%u\n", (unsigned int)BATCH_SIZE, (unsigned int)currentSeq);
                            Serial.write(txBuf, BATCH_SIZE);
                            Serial.flush();
                            ackWait = millis();
                            while (millis() - ackWait < 150) {
                                if (Serial.available()) {
                                    handleSerialCommands();
                                }
                                if (lastAudioAckSeq == currentSeq) {
                                    sendOk = true;
                                    break;
                                }
                                delay(2);
                            }
                        }
                        if (sendOk) {
                            totalStreamBytes += BATCH_SIZE;
                        } else if (fsReady) {
                            if (!isOfflineRecording) {
                                enforceFreeSpace();
                                isOfflineRecording = true;
                                char filename[40];
                                snprintf(filename, sizeof(filename), "/offline_inc_%lu.pcm", (unsigned long)millis());
                                offlineIncidentFile = LittleFS.open(filename, FILE_WRITE);
                            }
                            if (offlineIncidentFile) {
                                offlineIncidentFile.write(txBuf, BATCH_SIZE);
                            }
                        }
                    } else if (hasActiveStream && ws.isConnected() && serverAuthenticated && !isUploadingArchive) {
                        sendOk = ws.sendBIN(txBuf, BATCH_SIZE);
                    } else if (isOfflineRecording && offlineIncidentFile) {
                        sendOk = (offlineIncidentFile.write(txBuf, BATCH_SIZE) == BATCH_SIZE);
                    }

                    if (sendOk) {
                        ringTail = (ringTail + BATCH_SIZE) % RING_BUFFER_SIZE;
                        lastBatchSentAt = millis();
                    } else {
                        delay(4);
                    }
                }
            }
        }
    }

    // --- PIN KEEP-ALIVE & TELEMETRY HEARTBEAT ---
    // 1. Authoritative direct channel: ONLY when serverAuthenticated == true!
    static unsigned long lastHeartbeatMs = 0;
    if (ws.isConnected() && serverAuthenticated) {
        if (needImmediateHeartbeat || (millis() - lastHeartbeatMs >= 3000)) {
            needImmediateHeartbeat = false;
            lastHeartbeatMs = millis();
            int rssi = WiFi.RSSI();
            BatteryTelemetry bt = readBatteryTelemetry();
            ws.sendTXT("PING:" + String(rssi) + ":" + String(bt.batteryPct) + ":" + String(bt.batteryMv));
        }
    }

    processArchiveUpload();

    // --- THERMAL & POWER MANAGEMENT ---
    // In dual-core mode, audio sampling runs on Core 1. When not actively streaming,
    // yield Core 0 to the FreeRTOS idle task (waiti) for 10ms to drop CPU load from 100% to ~2%
    // and keep the ESP32 cool to the touch!
    if (!isStreaming && !isUploadingArchive) {
        delay(10);
    }
}
