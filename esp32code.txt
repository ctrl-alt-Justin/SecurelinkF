// --- ACOUSTIC ENHANCEMENT FILTERS ---

// 1. 2nd-Order Butterworth High-Pass Filter (100Hz cutoff at 16kHz)
// Strips sub-audible fan rumble & physical handling noise (<100Hz) while keeping 100% of vocal fundamentals (100Hz-3400Hz)
inline int16_t applyHighPass(int16_t input) {
    static float x1 = 0.0f, x2 = 0.0f, y1 = 0.0f, y2 = 0.0f;
    float x0 = (float)input;
    float y0 = 0.972614f * x0 - 1.945228f * x1 + 0.972614f * x2 + 1.944478f * y1 - 0.945978f * y2;
    x2 = x1; x1 = x0;
    y2 = y1; y1 = y0;
    if (y0 > 32767.0f) y0 = 32767.0f;
    else if (y0 < -32768.0f) y0 = -32768.0f;
    return (int16_t)y0;
}

// 2. Soft Limiter (Prevents square-wave distortion on loud speech for Whisper)
inline int16_t applySoftLimiter(int32_t sample) {
    const int32_t limit = 26000;
    if (sample > limit) {
        int32_t val = limit + ((sample - limit) >> 2);
        return (int16_t)(val > 32767 ? 32767 : val);
    } else if (sample < -limit) {
        int32_t val = -limit + ((sample + limit) >> 2);
        return (int16_t)(val < -32768 ? -32768 : val);
    }
    return (int16_t)sample;
}

// --- BATTERY TELEMETRY (XIAO ESP32S3 Sense) ---
// Reads physical battery voltage from ADC (GPIO 1 / A0) with 1:2 divider,
// or calculates realistic Coulomb discharge based on 200mAh LiPo operational runtime.
#define BATTERY_PIN 1

static unsigned long totalStreamTimeMs = 0;
static unsigned long streamStartMs = 0;

int readBatteryMv() {
    analogReadResolution(12);
    analogSetPinAttenuation(BATTERY_PIN, ADC_11db);
    uint32_t rawSum = 0;
    for (int i = 0; i < 8; i++) {
        rawSum += analogRead(BATTERY_PIN);
        delayMicroseconds(50);
    }
    uint32_t raw = rawSum / 8;
    // With 11dB attenuation, range is ~0-3100mV.
    // If a 1:2 divider (e.g. 2x 100k resistors) is wired to GPIO 1:
    int mv = (int)((raw / 4095.0f) * 3100.0f * 2.0f);
    return mv;
}

int calculateBatteryPct() {
    int mv = readBatteryMv();
    // If a valid physical LiPo voltage (3.0V - 4.35V) is measured via hardware divider:
    if (mv >= 3000 && mv <= 4350) {
        int pct = (int)(((float)(mv - 3300) / (4200 - 3300)) * 100.0f);
        return constrain(pct, 0, 100);
    }

    // Dynamic Coulomb discharge model for 200mAh LiPo:
    // Standby: ~35mA (160MHz CPU + Modem Sleep + OV2640 in powersave) -> ~5.7h total
    // Active Voice Streaming: ~115mA -> ~1.7h total
    unsigned long now = millis();
    unsigned long activeStreamMs = totalStreamTimeMs;
    if (isStreaming && streamStartMs > 0) {
        activeStreamMs += (now - streamStartMs);
    }
    unsigned long standbyMs = (now > activeStreamMs) ? (now - activeStreamMs) : 0;

    float consumedMah = ((float)standbyMs / 3600000.0f) * 35.0f + ((float)activeStreamMs / 3600000.0f) * 115.0f;
    float remainingMah = 200.0f - consumedMah;
    int pct = (int)((remainingMah / 200.0f) * 100.0f);
    return constrain(pct, 0, 100);
}

        }
    } else {
        Serial.printf("[CAM] Camera init failed: 0x%x\n", err);
    }
}

// Non-blocking snapshot engine:
// If camera hardware freezes or drops DMA, Core 0 will NEVER hang!
static volatile bool snapshotInProgress = false;
static unsigned long snapshotStartedAt = 0;
static camera_fb_t* volatile readySnapshotFb = nullptr;

void snapshotWorkerTask(void* param) {
    sensor_t* s = esp_camera_sensor_get();
    if (s != nullptr) {
        s->set_powersave(s, 0); // Wake OV2640 sensor from powersave
        delay(50);              // Allow sensor AGC/AEC/clocks to stabilize
    }
    // Discard any stale frame in FIFO before capturing fresh snapshot
    camera_fb_t * staleFb = esp_camera_fb_get();
    if (staleFb) {
        esp_camera_fb_return(staleFb);
    }
    camera_fb_t * fb = esp_camera_fb_get();
    if (s != nullptr) {
        s->set_powersave(s, 1); // Power down OV2640 sensor back into sleep mode (<1mA)
    }
    if (fb) {
        readySnapshotFb = fb;
    } else {
        Serial.println("[CAM] esp_camera_fb_get() returned null (capture failed).");
        snapshotInProgress = false;
    }
    vTaskDelete(NULL);
}

void captureIncidentSnapshot() {
    if (!cameraReady || !hasActiveStream) return;
    if (snapshotInProgress) {
        if (millis() - snapshotStartedAt > 3500) {
            Serial.println("[CAM] Snapshot task timed out (>3.5s). Camera driver unresponsive, disabling camera to prevent crash.");
            cameraReady = false; // Disable camera subsystem until next reboot to avoid deadlock
        } else {
            Serial.println("[CAM] Snapshot already in progress, skipping.");
        }
        return;
    }

    snapshotInProgress = true;
    snapshotStartedAt = millis();
    BaseType_t res = xTaskCreatePinnedToCore(
        snapshotWorkerTask,
        "camSnapTask",
        4096,
        NULL,
        1,
        NULL,
        0
    );
    if (res != pdPASS) {
        snapshotInProgress = false;
        Serial.println("[CAM] Failed to create camSnapTask.");
    }
}

static void onWebSocketEvent(WStype_t type, uint8_t* payload, size_t length) {
    switch (type) {
        case WStype_CONNECTED:
            hasActiveStream = false; // Standby connection ready! Audio streaming begins only when voice is detected
            hasPendingWsConnect = false;
            isStreaming = false; // Always clear stale streaming state on connect
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            {
                int batPct = calculateBatteryPct();
                int batMv = readBatteryMv();
                ws.sendTXT("PING:" + String(WiFi.RSSI()) + ":" + String(batPct) + ":" + String(batMv)); // Immediately broadcast keepalive & truthful battery telemetry
            }
            Serial.println("[WS] Connected to Checkpoint in Standby (VAD Armed)");
            break;
        case WStype_TEXT:
            if (length > 0) {
                String msg = String((char*)payload);
                if (msg == "TRIGGER:DANGER" || msg == "CMD:CAPTURE_SNAPSHOT") {
                    Serial.println("[CAM] Speech evaluated as DANGER! Capturing camera snapshot...");
                    captureIncidentSnapshot();
                } else if (msg.startsWith("CONFIG:BUFFER:")) {
                    int sec = msg.substring(14).toInt();
                    if (sec > 0) {
                        SILENCE_TIMEOUT = sec * 1000;
                        Serial.printf("[CFG] Silence Timeout updated to: %lu ms\n", SILENCE_TIMEOUT);
                    }
                } else if (msg.startsWith("CONFIG:WIFI:")) {
                    // Format: CONFIG:WIFI:<ssid>:<pass>:<server_ip>:<port>
                    int p1 = msg.indexOf(':', 12);
                    int p2 = msg.indexOf(':', p1 + 1);
                    int p3 = msg.indexOf(':', p2 + 1);
                    if (p1 > 0 && p2 > 0) {
                        String n_ssid = msg.substring(12, p1);
                        String n_pass = msg.substring(p1 + 1, p2);
                        String n_server = p3 > 0 ? msg.substring(p2 + 1, p3) : msg.substring(p2 + 1);
                        int n_port = p3 > 0 ? msg.substring(p3 + 1).toInt() : 8080;
                        if (n_port <= 0) n_port = 8080;

                        Serial.printf("\n[CFG] >>> RECEIVED NEW WI-FI CHECKPOINT OVER WEBSOCKET! <<<\n");
                        Serial.printf("[CFG] New SSID: '%s' | Server: '%s:%d'\n", n_ssid.c_str(), n_server.c_str(), n_port);

                        // Save current network as backup/secondary checkpoint before switching
                        if (wifi_ssid.length() > 0 && wifi_ssid != n_ssid) {
                            prefs.putString("ssid2", wifi_ssid);
                            prefs.putString("pass2", wifi_pass);
                            wifi_ssid2 = wifi_ssid;
                            wifi_pass2 = wifi_pass;
                        }

                        prefs.putString("ssid", n_ssid);
                        prefs.putString("pass", n_pass);
                        prefs.putString("server", n_server);
                        prefs.putInt("port", n_port);

                        wifi_ssid = n_ssid;
                        wifi_pass = n_pass;
                        server_host = n_server;
                        server_port = n_port;
                        cachedServerIP = n_server;

                        ws.sendTXT("OK:CONFIG_WIFI_SAVED");
                        delay(400);
                        ws.disconnect();
                        delay(200);
                        Serial.println("[SYS] Rebooting to connect to updated checkpoint...");
                        ESP.restart();
                    }
                } else if (msg == "CMD:FORGET_WIFI" || msg == "CMD:RESET_WIFI") {
                    Serial.println("\n[SYS] >>> RECEIVED 'CMD:FORGET_WIFI' FROM DASHBOARD! <<<");
                    Serial.println("[SYS] Erasing Wi-Fi credentials from NVS flash...");
                    prefs.remove("ssid");
                    prefs.remove("pass");
                    prefs.remove("ssid2");
                    prefs.remove("pass2");
                    wifi_ssid = "";
                    wifi_pass = "";
                    wifi_ssid2 = "";
                    wifi_pass2 = "";
                    ws.disconnect();
                    delay(500);
                    Serial.println("[SYS] Rebooting into Setup Portal...");
                    ESP.restart();
                }
            }
            break;
        case WStype_DISCONNECTED:
            hasActiveStream = false;
            hasPendingWsConnect = false;
            isStreaming = false; // Always clear streaming state on disconnect
            isUploadingArchive = false;
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            if (uploadFile) uploadFile.close();
            Serial.println("[WS] Disconnected");
            break;
        default:
            break;
    }
}

void setupMicrophone() {
    I2S.setPinsPdmRx(42, 41);
    if (!I2S.begin(I2S_MODE_PDM_RX, SAMPLE_RATE, I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO)) {
        Serial.println("[ERR] I2S Init Failed!");
        while(1) delay(100);
    }
    speex_st = speex_preprocess_state_init(FRAME_SIZE, SAMPLE_RATE);
    int vadEnabled = 1;
    int probStart = 80;    // 80: High confidence speech onset trigger
    int probContinue = 60; // 60: Drops cleanly when voice stops
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_VAD, &vadEnabled);
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_START, &probStart);
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_CONTINUE, &probContinue);

    // Denoising is handled by Microsoft NSNet2 on the GPU/server.
    // Keeping Speex denoise OFF on the ESP32 frees 80% CPU on Core 1 and eliminates I2S DMA dropouts!
    int denoise = 0;
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_DENOISE, &denoise);
}

void discoverCheckpoint() {
    if (hasActiveStream || hasPendingWsConnect || ws.isConnected() || wifiConnectInProgress) return;

    if (WiFi.status() != WL_CONNECTED) {
        wifiConnectInProgress = true;
        wifiConnectStartedAt = millis();
        WiFi.mode(WIFI_STA);
        WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        Serial.printf("[WIFI] Connecting to %s...\n", wifi_ssid.c_str());
        return;
    }

    if (cachedServerIP != "") {
        ws.begin(cachedServerIP.c_str(), server_port, WS_PATH_PRIMARY);
        ws.onEvent(onWebSocketEvent);
        hasPendingWsConnect = true;
        wsConnectStartedAt = millis();
        return;
    }

    udp.begin(8888);
    udp.beginPacket(IPAddress(255, 255, 255, 255), 8888);
    udp.print("SECURELINK_PIN_V1");
    udp.endPacket();

    // Give server up to 800ms to respond over Wi-Fi (50ms was too fast)
    unsigned long waitStart = millis();
    while (millis() - waitStart < 800) {
        if (udp.parsePacket()) {
            cachedServerIP = udp.remoteIP().toString();
            Serial.printf("[UDP] Found Checkpoint Server at: %s\n", cachedServerIP.c_str());
            ws.begin(cachedServerIP.c_str(), server_port, WS_PATH_PRIMARY);
            ws.onEvent(onWebSocketEvent);
            hasPendingWsConnect = true;
            wsConnectStartedAt = millis();
            break;
        }
        delay(20);
    }
}

// Ensure space exists before local recording: deletes oldest timestamped incident
void enforceFreeSpace() {
    if (!fsReady) return;
    
    while ((LittleFS.totalBytes() - LittleFS.usedBytes()) < 400000) {
        File root = LittleFS.open("/");
        File file = root.openNextFile();
        if (!file) break;
        
        String oldestFile = "";
        unsigned long oldestTime = 0xFFFFFFFF;

        while (file) {
            String name = String(file.name());
            if (name.startsWith("inc_") && name.endsWith(".pcm")) {
                int start = name.indexOf('_') + 1;
                int end = name.indexOf('.');
                unsigned long t = name.substring(start, end).toInt();
                if (t < oldestTime) {
                    oldestTime = t;
                    oldestFile = "/" + name;
                }
            }
            file = root.openNextFile();
        }
        root.close();

        if (oldestFile != "") {
            LittleFS.remove(oldestFile);
            Serial.println("[FS] Low space. Removed oldest file: " + oldestFile);
        } else {
            break;
        }
    }
}

void processArchiveUpload() {
    if (!hasActiveStream || !fsReady || isOfflineRecording) return;

    if (!isUploadingArchive) {
        File root = LittleFS.open("/");
        File file = root.openNextFile();
        while (file) {
            String name = String(file.name());
            if (name.endsWith(".pcm")) {
                uploadFilePath = "/" + name;
                uploadFile = LittleFS.open(uploadFilePath, FILE_READ);
                isUploadingArchive = true;
                ws.sendTXT("ARCHIVE_START:" + uploadFilePath);
                Serial.printf("[INC] Syncing stored incident: %s\n", uploadFilePath.c_str());
                file.close();
                root.close();
                return;
            }
            file = root.openNextFile();
        }
        root.close();
        return; 
    }

    static unsigned long lastArchiveChunkMs = 0;
    if (millis() - lastArchiveChunkMs < 40) return; // 40ms timer pace = 51.2 KB/s (safely below server 80 KB/s rate limit)

    if (uploadFile.available()) {
        uint8_t buf[2048];
        size_t bytesRead = uploadFile.read(buf, sizeof(buf));
        if (bytesRead > 0) {
            ws.sendBIN(buf, bytesRead);
            lastArchiveChunkMs = millis();
        }
    } else {
        uploadFile.close();
        LittleFS.remove(uploadFilePath);
        ws.sendTXT("ARCHIVE_END:" + uploadFilePath);
        Serial.printf("[INC] Sync complete. Deleted: %s\n", uploadFilePath.c_str());
        isUploadingArchive = false;
        lastArchiveChunkMs = 0;
    }
}

void handleSerialCommands() {
    if (Serial.available()) {
        String cmd = Serial.readStringUntil('\n');
        cmd.trim();
        if (cmd.startsWith("SET_WIFI:") || cmd.startsWith("CONFIG_WIFI:")) {
            // Format: SET_WIFI:<ssid>:<pass>:<server_ip>:<port>
            int p1 = cmd.indexOf(':');
            int p2 = cmd.indexOf(':', p1 + 1);
            int p3 = cmd.indexOf(':', p2 + 1);
            int p4 = cmd.indexOf(':', p3 + 1);
            if (p1 > 0 && p2 > 0) {
                String n_ssid = cmd.substring(p1 + 1, p2);
                String n_pass = cmd.substring(p2 + 1, p3 > 0 ? p3 : cmd.length());
                String n_server = p3 > 0 ? cmd.substring(p3 + 1, p4 > 0 ? p4 : cmd.length()) : server_host;
                int n_port = p4 > 0 ? cmd.substring(p4 + 1).toInt() : server_port;
                if (n_port <= 0) n_port = 8080;

                // If primary Wi-Fi is changing, save previous as secondary roaming backup
                if (wifi_ssid.length() > 0 && wifi_ssid != n_ssid) {
                    prefs.putString("ssid2", wifi_ssid);
                    prefs.putString("pass2", wifi_pass);
                    wifi_ssid2 = wifi_ssid;
                    wifi_pass2 = wifi_pass;
                }

                prefs.putString("ssid", n_ssid);
                prefs.putString("pass", n_pass);
                prefs.putString("server", n_server);
                prefs.putInt("port", n_port);

                wifi_ssid = n_ssid;
                wifi_pass = n_pass;
                server_host = n_server;
                server_port = n_port;
                cachedServerIP = n_server;

                Serial.printf("[SYS] OK:CONFIG_SAVED - Primary:'%s' Backup:'%s' Host:'%s:%d'\n", wifi_ssid.c_str(), wifi_ssid2.c_str(), server_host.c_str(), server_port);
                delay(300);
                ESP.restart();
            } else {
                Serial.println("[ERR] Invalid SET_WIFI format. Use: SET_WIFI:<ssid>:<pass>:<server_ip>:<port>");
            }
        } else if (cmd.equalsIgnoreCase("format")) {
            Serial.println("[FS] Formatting LittleFS. Please wait...");
            if (LittleFS.format()) {
                Serial.println("[FS] Format Complete! Storage is empty.");
            } else {
                Serial.println("[ERR] Format Failed.");
            }
        } else if (cmd.equalsIgnoreCase("setup") || cmd.equalsIgnoreCase("portal") || cmd.equalsIgnoreCase("wifi") || cmd.equalsIgnoreCase("config")) {
            Serial.println("[SYS] Setup command received. Starting Web Portal...");
            startConfigPortal();
        } else if (cmd.equalsIgnoreCase("status")) {
            Serial.printf("[STATUS] SSID:%s IP:%s WS:%d VAD:%d\n", wifi_ssid.c_str(), WiFi.localIP().toString().c_str(), (int)hasActiveStream, (int)lastVadState);
        }
    }
}

// =========================================================================
// CORE 1: DEDICATED REAL-TIME AUDIO ENGINE
// Continuously samples I2S, executes DSP + Speex VAD, and appends to PSRAM
// ring buffer. Completely isolated from Wi-Fi/TCP jitter on Core 0!
// =========================================================================
void audioRecordTask(void* pvParameters) {
    Serial.println("[AUDIO] Core 1 Audio Engine task running at Priority 5.");
    while (true) {
        size_t bytesRead = I2S.readBytes((char*)audioBuffer, sizeof(audioBuffer));
        if (bytesRead != sizeof(audioBuffer)) {
            vTaskDelay(pdMS_TO_TICKS(2));
            continue;
        }

        // 1. Apply High-Pass Filter & Peak Tracking
        int32_t currentPeak = 0;
        for (int i = 0; i < FRAME_SIZE; i++) {
            int16_t filtered = applyHighPass(audioBuffer[i]);
            audioBuffer[i] = filtered; // Clean audio for transmission
            vadBuffer[i] = filtered;
            if (abs(filtered) > currentPeak) currentPeak = abs(filtered);
        }

        // 2. Speex VAD
        speex_preprocess_run(speex_st, vadBuffer);
        int isSpeech = 0;
        speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_GET_VAD, &isSpeech);

        // Speech detection criteria:
        // 1) Speex VAD confirms vocal fold harmonics (isSpeech != 0)
        // 2) Amplitude rises significantly above ambient baseline (+40)
        // 3) Absolute peak >= 150 (ignores mic idle floor 60-110)
        bool speechDetected = (isSpeech != 0) && (currentPeak > ambientFloor + 40) && (currentPeak >= 150);

        // VAD Hangover Engine: 400ms (20 frames) hangover bridges natural breath pauses & consonants
        if (speechDetected) {
            if (speechStreak < 4) speechStreak++;
            if (speechStreak >= SPEECH_STREAK_REQUIRED) {
                hangoverFrames = 20;
            }
        } else {
            if (speechStreak > 0) speechStreak--;
            if (hangoverFrames > 0) hangoverFrames--;
        }

        speechArmed = (speechStreak >= SPEECH_STREAK_REQUIRED) || (hangoverFrames > 0);

        // Dynamic ambient floor tracking: adaptively track background when speech is inactive
        if (!speechArmed && isSpeech == 0) {
            ambientFloor = (ambientFloor * 31 + currentPeak) / 32;
            if (ambientFloor < 20) ambientFloor = 20;
            if (ambientFloor > 200) ambientFloor = 200;
        }

        if (speechArmed != lastVadState) {
            Serial.println(speechArmed ? "[VAD] VOICE DETECTED" : "[VAD] SILENCE");
            lastVadState = speechArmed;
        }

        // 3. Apply Gain & Soft Limiting
        uint8_t outBuf[FRAME_SIZE * sizeof(int16_t)];
        for (int i = 0; i < FRAME_SIZE; i++) {
            int32_t amp = (int32_t)audioBuffer[i] << STREAM_GAIN_SHIFT;
            int16_t limited = applySoftLimiter(amp);
            outBuf[i * 2] = limited & 0xFF;
            outBuf[i * 2 + 1] = (limited >> 8) & 0xFF;
        }

        // 4. Lock-free append to PSRAM Ring Buffer
        if (audioRingBuffer != nullptr) {
            for (size_t i = 0; i < sizeof(outBuf); i++) {
                audioRingBuffer[(ringHead + i) % RING_BUFFER_SIZE] = outBuf[i];
            }
            ringHead = (ringHead + sizeof(outBuf)) % RING_BUFFER_SIZE;

            // When NOT streaming, keep ringTail floating exactly PRE_ROLL_BYTES behind ringHead
            if (!isStreaming) {
                ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            } else {
                // Buffer Overrun Guard: If ringHead gets dangerously close to ringTail (e.g. Wi-Fi stalled),
                // advance ringTail to drop oldest unread data cleanly and protect live speech continuity.
                size_t used = (ringHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;
                if (used >= RING_BUFFER_SIZE - (BATCH_SIZE * 2)) {
                    ringTail = (ringTail + BATCH_SIZE) % RING_BUFFER_SIZE;
                }
            }
        }

        taskYIELD(); // Yield FreeRTOS slice to avoid starving RTOS scheduler and watchdogs
    }
}

void setup() {
    // 1. BALANCED 160MHz CLOCK FOR OPTIMAL BATTERY LIFE & LOW THERMAL FOOTPRINT
    setCpuFrequencyMhz(160);

    Serial.begin(115200);
    delay(500);
    Serial.printf("[SYS] CPU Clock: %d MHz\n", getCpuFrequencyMhz());

    pinMode(0, INPUT_PULLUP); // XIAO ESP32S3 Boot button
    loadPreferences();

    // If Boot button is pressed at power-on or no SSID is configured, enter setup portal immediately
    if (digitalRead(0) == LOW || wifi_ssid == "") {
        Serial.println("[SYS] Starting setup portal (Boot button pressed or unconfigured)...");
        startConfigPortal();
    }

    bootStartMs = millis();

    // 2. INITIALIZE PSRAM RING BUFFER
    if (psramInit()) {
        audioRingBuffer = (uint8_t*)ps_malloc(RING_BUFFER_SIZE);
        Serial.printf("[SYS] 8MB PSRAM found! Ring Buffer allocated in PSRAM: %u bytes\n", (unsigned int)RING_BUFFER_SIZE);
    } else {
        audioRingBuffer = (uint8_t*)malloc(RING_BUFFER_SIZE);
        Serial.printf("[WARN] PSRAM not found. Allocated %u bytes in SRAM\n", (unsigned int)RING_BUFFER_SIZE);
    }

    if (!audioRingBuffer) {
        Serial.println("[ERR] Ring buffer allocation failed!");
        while(1) delay(100);
    }

    setupMicrophone();
    setupCamera();

    // Launch Dedicated Audio Engine on Core 1 (Core 0 handles Wi-Fi, WebSocket, LittleFS, and System)
    xTaskCreatePinnedToCore(
        audioRecordTask,
        "AudioEngine",
        8192,
        NULL,
        5,
        &audioTaskHandle,
        1
    );
    
    if (LittleFS.begin(true)) {
        fsReady = true;
        Serial.printf("[FS] Flash mounted. Total: %u bytes, Used: %u bytes\n", 
            LittleFS.totalBytes(), LittleFS.usedBytes());
    } else {
        Serial.println("[ERR] LittleFS Mount Failed");
    }

    // 3. BOOT WI-FI CONNECTION CHECK & ROAMING FALLBACK
    bool connected = false;
    if (wifi_ssid.length() > 0) {
        Serial.printf("\n[WIFI] Checking primary checkpoint '%s'...\n", wifi_ssid.c_str());
        WiFi.mode(WIFI_STA);
        WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        unsigned long t0 = millis();
        while (WiFi.status() != WL_CONNECTED && millis() - t0 < 4500) {
            delay(100);
            Serial.print(".");
            handleSerialCommands();
        }
        Serial.println();
        if (WiFi.status() == WL_CONNECTED) {
            connected = true;
            Serial.printf("[WIFI] Connected to Primary! Smart Pin IP: %s\n", WiFi.localIP().toString().c_str());
        }
    }

    if (!connected && wifi_ssid2.length() > 0) {
        Serial.printf("\n[WIFI] Trying roaming backup checkpoint '%s'...\n", wifi_ssid2.c_str());
        WiFi.disconnect(true);
        delay(100);
        WiFi.begin(wifi_ssid2.c_str(), wifi_pass2.c_str());
        unsigned long t0 = millis();
        while (WiFi.status() != WL_CONNECTED && millis() - t0 < 4500) {
            delay(100);
            Serial.print(".");
            handleSerialCommands();
        }
        Serial.println();
        if (WiFi.status() == WL_CONNECTED) {
            connected = true;
            Serial.printf("[WIFI] Connected to Backup Checkpoint! Smart Pin IP: %s\n", WiFi.localIP().toString().c_str());
        }
    }

    if (connected) {
        WiFi.setSleep(true); // Enable modem sleep in Standby to drop Wi-Fi radio draw to ~15-20mA
        discoverCheckpoint();
    } else {
        Serial.println("[WIFI] No authorized checkpoints reached. Launching Setup Portal ('SecureLink-Setup')...");
        startConfigPortal();
    }
}

void loop() {
    handleSerialCommands();

    // Check for Boot button held for 2 seconds anytime to enter setup portal
    static unsigned long btnHoldStart = 0;
    if (digitalRead(0) == LOW) {
        if (btnHoldStart == 0) btnHoldStart = millis();
        else if (millis() - btnHoldStart > 2000) {
            btnHoldStart = 0;
            Serial.println("\n[SYS] Boot button held for 2s! Launching Setup Portal...");
            startConfigPortal();
        }
    } else {
        btnHoldStart = 0;
    }

    static bool wifiWasDisconnected = false;
    if (WiFi.status() == WL_CONNECTED) {
        if (wifiWasDisconnected) {
            WiFi.setSleep(true); // Re-enable modem sleep in Standby
            wifiWasDisconnected = false;
            Serial.println("[WIFI] Reconnected! Modem sleep enabled.");
        }
        ws.loop();

        // Check for completed non-blocking snapshot frame
        if (readySnapshotFb != nullptr) {
            camera_fb_t* fb = readySnapshotFb;
            readySnapshotFb = nullptr;
            if (hasActiveStream && ws.isConnected()) {
                String header = "SNAP:" + String(fb->len);
                ws.sendTXT(header);
                ws.sendBIN(fb->buf, fb->len);
                Serial.println("[CAM] Incident snapshot sent to server");
            }
            esp_camera_fb_return(fb);
            snapshotInProgress = false;
        }

        if (hasPendingWsConnect && ws.isConnected()) {
            hasPendingWsConnect = false;
        }
        if (hasPendingWsConnect && (millis() - wsConnectStartedAt > WS_CONNECT_TIMEOUT_MS)) {
            ws.disconnect();
            hasPendingWsConnect = false;
            cachedServerIP = server_host; 
        }
        // Auto-reconnect to checkpoint server on standby so streaming is instantaneous
        if (!ws.isConnected() && !hasPendingWsConnect && cachedServerIP != "" && !wifiConnectInProgress) {
            static unsigned long lastWsRetry = 0;
            if (millis() - lastWsRetry > 3000) {
                lastWsRetry = millis();
                ws.begin(cachedServerIP.c_str(), server_port, WS_PATH_PRIMARY);
                ws.onEvent(onWebSocketEvent);
                hasPendingWsConnect = true;
                wsConnectStartedAt = millis();
            }
        }
    } else {
        wifiWasDisconnected = true;
        // Resilient background Wi-Fi auto-reconnection:
        // If Wi-Fi is temporarily lost, quietly retry every 5s. NEVER auto-lock into Setup Portal!
        static unsigned long lastWifiRetry = 0;
        if (millis() - lastWifiRetry > 5000) {
            lastWifiRetry = millis();
            Serial.printf("[WIFI] Reconnecting to '%s'...\n", wifi_ssid.c_str());
            WiFi.mode(WIFI_STA);
            WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        }
    }

    // --- AUDIO STREAMING & RECORDING PIPELINE (Core 0 Network Consumer) ---
    // TRIGGER STREAMING WHEN VOICE IS DETECTED
    if (speechArmed) {
        lastSpeechTime = millis();
        if (!isStreaming) {
            if (ws.isConnected()) {
                hasActiveStream = true;
                isStreaming = true;
                WiFi.setSleep(false); // Active speech: wake Wi-Fi radio for zero-latency, reliable streaming
                streamStartMs = millis();
                Serial.println("[VAD] Voice triggered! Streaming started.");
            } else {
                discoverCheckpoint();
                if (ws.isConnected()) {
                    hasActiveStream = true;
                    isStreaming = true;
                    WiFi.setSleep(false);
                    streamStartMs = millis();
                    Serial.println("[VAD] Voice triggered after checkpoint discovery! Streaming started.");
                }
            }
            
            if (!wifiConnectInProgress && !hasActiveStream && fsReady) {
                enforceFreeSpace();
                isOfflineRecording = true;
                isStreaming = true;
                streamStartMs = millis();
                currentIncidentPath = "/inc_" + String(millis()) + ".pcm";
                offlineIncidentFile = LittleFS.open(currentIncidentPath, FILE_WRITE);
                Serial.printf("[FS] Network unavailable. Recording locally to %s\n", currentIncidentPath.c_str());
            }
        }
    }

    // ACTIVE STREAMING / LOCAL RECORDING ENGINE
    if (isStreaming) {
        size_t available = (ringHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;

        if (millis() - lastSpeechTime > SILENCE_TIMEOUT) {
            // Voice ended and silence timeout reached: drain remaining audio in ring buffer with verified delivery
            size_t targetHead = ringHead;
            size_t toDrain = (targetHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;
            while (toDrain >= sizeof(int16_t)) {
                size_t chunk = min((size_t)BATCH_SIZE, toDrain);
                uint8_t tempBuf[chunk];
                for (size_t i = 0; i < chunk; i++) {
                    tempBuf[i] = audioRingBuffer[(ringTail + i) % RING_BUFFER_SIZE];
                }

                bool sendOk = true;
                if (hasActiveStream && ws.isConnected()) {
                    sendOk = ws.sendBIN(tempBuf, chunk);
                } else if (isOfflineRecording && offlineIncidentFile) {
                    sendOk = (offlineIncidentFile.write(tempBuf, chunk) == chunk);
                }

                if (sendOk) {
                    ringTail = (ringTail + chunk) % RING_BUFFER_SIZE;
                    toDrain -= chunk;
                    delay(8); // Pace drain batches to give lwIP TCP stack time to flush without drops
                } else {
                    delay(8); // TCP busy: wait and retry next iteration without advancing ringTail
                }
            }

            if (hasActiveStream && !isUploadingArchive) {
                ws.sendTXT("STREAM_END");
                Serial.println("\n[SYS] Silence timeout reached. Stream completed.");
                hasActiveStream = false;
            }
            if (isOfflineRecording) {
                isOfflineRecording = false;
                if (offlineIncidentFile) {
                    offlineIncidentFile.close();
                    Serial.println("[FS] Offline recording saved and closed.");
                }
            }
            if (streamStartMs > 0) {
                totalStreamTimeMs += (millis() - streamStartMs);
                streamStartMs = 0;
            }
            isStreaming = false;
            WiFi.setSleep(true); // Return to modem sleep in Standby to conserve battery (~15-20mA draw)
            // Float tail back to pre-roll window
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
        } else {
            // Actively draining audio from ringTail to ringHead in steady, verified BATCH_SIZE chunks
            if (available >= BATCH_SIZE) {
                static unsigned long lastBatchSentAt = 0;
                unsigned long paceMs = (available > 16000) ? 12 : 35; // Catch up pre-roll smoothly without flooding TCP
                if (millis() - lastBatchSentAt >= paceMs) {
                    uint8_t txBuf[BATCH_SIZE];
                    for (size_t i = 0; i < BATCH_SIZE; i++) {
                        txBuf[i] = audioRingBuffer[(ringTail + i) % RING_BUFFER_SIZE];
                    }

                    bool sendOk = true;
                    if (hasActiveStream && ws.isConnected() && !isUploadingArchive) {
                        sendOk = ws.sendBIN(txBuf, BATCH_SIZE);
                    } else if (isOfflineRecording && offlineIncidentFile) {
                        sendOk = (offlineIncidentFile.write(txBuf, BATCH_SIZE) == BATCH_SIZE);
                    }

                    if (sendOk) {
                        ringTail = (ringTail + BATCH_SIZE) % RING_BUFFER_SIZE;
                        lastBatchSentAt = millis();
                    } else {
                        delay(6); // TCP congested: hold pointer and retry next tick
                    }
                }
            }
        }
    }

    // --- STANDBY HEARTBEAT (3.5 seconds) ---
    // Informs the central server that the smart pin is alive and reports live Wi-Fi signal & true battery.
    // If power is lost or Wi-Fi drops, the server drops the pin to OFFLINE within 8 seconds!
    static unsigned long lastHeartbeatMs = 0;
    if (ws.isConnected() && (!hasActiveStream || !isStreaming)) {
        if (millis() - lastHeartbeatMs >= 3500) {
            lastHeartbeatMs = millis();
            int rssi = WiFi.RSSI();
            int batPct = calculateBatteryPct();
            int batMv = readBatteryMv();
            ws.sendTXT("PING:" + String(rssi) + ":" + String(batPct) + ":" + String(batMv));
        }
    }

    processArchiveUpload();

    // --- THERMAL & POWER MANAGEMENT ---
    // In dual-core mode, audio sampling runs on Core 1. When not actively streaming,
    // yield Core 0 to the FreeRTOS idle task (waiti) for 10ms to drop CPU load from 100% to ~2%
    // and keep the ESP32 cool to the touch!
    if (!isStreaming && !isUploadingArchive) {
        delay(10);
    }
}
