// Lightweight JSON field extraction utility for memory-constrained ESP32
String extractJsonField(const String& json, const String& key) {
    int kIdx = json.indexOf("\"" + key + "\"");
    if (kIdx < 0) return "";
    int colIdx = json.indexOf(":", kIdx);
    if (colIdx < 0) return "";
    int startQuote = json.indexOf("\"", colIdx);
    if (startQuote < 0) {
        int endComma = json.indexOf(",", colIdx);
        int endBrace = json.indexOf("}", colIdx);
        int end = (endComma > 0 && endComma < endBrace) ? endComma : endBrace;
        if (end < 0) end = json.length();
        String val = json.substring(colIdx + 1, end);
        val.trim();
        return val;
    }
    int endQuote = json.indexOf("\"", startQuote + 1);
    if (endQuote < 0) return "";
    return json.substring(startQuote + 1, endQuote);
}

void connectWebSocket(const String& host, int port) {
    String cleanHost = host;
    bool isSsl = false;
    if (cleanHost.startsWith("wss://")) {
        cleanHost = cleanHost.substring(6);
        isSsl = true;
    } else if (cleanHost.startsWith("ws://")) {
        cleanHost = cleanHost.substring(5);
    } else if (cleanHost.startsWith("https://")) {
        cleanHost = cleanHost.substring(8);
        isSsl = true;
    } else if (cleanHost.startsWith("http://")) {
        cleanHost = cleanHost.substring(7);
    }
    if (cleanHost.indexOf('/') >= 0) cleanHost = cleanHost.substring(0, cleanHost.indexOf('/'));
    if (cleanHost.indexOf(':') >= 0) cleanHost = cleanHost.substring(0, cleanHost.indexOf(':'));
    
    // Construct authenticated WebSocket query path (?device_id=PIN-XXXX&token=...)
    String wsPath = String(WS_PATH_PRIMARY) + "?device_id=" + pin_device_id;
    if (pin_device_token.length() > 0) {
        wsPath += "&token=" + pin_device_token;
    }

    if (cleanHost.indexOf("trycloudflare.com") >= 0 || port == 443 || isSsl) {
        Serial.printf("[WS] Connecting via WSS/SSL to %s:443%s [token=%s]\n", cleanHost.c_str(), WS_PATH_PRIMARY, pin_device_token.length() > 0 ? "AUTHENTICATED" : "NONE");
        ws.beginSSL(cleanHost.c_str(), 443, wsPath.c_str());
    } else {
        Serial.printf("[WS] Connecting via WS to %s:%d%s [token=%s]\n", cleanHost.c_str(), port, WS_PATH_PRIMARY, pin_device_token.length() > 0 ? "AUTHENTICATED" : "NONE");
        ws.begin(cleanHost.c_str(), port, wsPath.c_str());
    }
    ws.onEvent(onWebSocketEvent);
    hasPendingWsConnect = true;
    wsConnectStartedAt = millis();
}

void sendBleText(const String& txt) {
    if (!pTxCharacteristic || !bleClientConnected) return;
    if (bleTxMutex != nullptr) {
        if (xSemaphoreTake(bleTxMutex, pdMS_TO_TICKS(100)) != pdTRUE) {
            return;
        }
    }
    pTxCharacteristic->setValue((uint8_t*)txt.c_str(), txt.length());
    pTxCharacteristic->notify();
    if (bleTxMutex != nullptr) {
        xSemaphoreGive(bleTxMutex);
    }
}

bool isCompleteJsonObject(const String& str) {
    if (str.length() < 2) return false;
    if (!str.startsWith("{") || !str.endsWith("}")) return false;
    int depth = 0;
    bool inQuotes = false;
    bool escape = false;
    for (size_t i = 0; i < str.length(); i++) {
        char c = str.charAt(i);
        if (escape) {
            escape = false;
            continue;
        }
        if (c == '\\') {
            escape = true;
            continue;
        }
        if (c == '\"') {
            inQuotes = !inQuotes;
            continue;
        }
        if (!inQuotes) {
            if (c == '{') depth++;
            else if (c == '}') {
                depth--;
                if (depth == 0 && i < str.length() - 1) {
                    return false;
                }
            }
        }
    }
    return (depth == 0);
}

class BleRxCallbacks: public BLECharacteristicCallbacks {
    void onWrite(BLECharacteristic *pCharacteristic) override {
        String chunk = pCharacteristic->getValue().c_str();
        if (chunk.length() == 0) return;

        unsigned long now = millis();
        // Discard stale partial fragments if idle for > 4 seconds
        if (bleRxBuffer.length() > 0 && (now - lastBleRxChunkTime > 4000)) {
            Serial.println("[BLE-RX] Clearing stale partial buffer after timeout.");
            bleRxBuffer = "";
        }
        lastBleRxChunkTime = now;

        Serial.printf("[BLE-RX] Received fragment (%d bytes)\n", chunk.length());

        if (bleRxBuffer.length() + chunk.length() > MAX_BLE_RX_BUFFER) {
            Serial.println("[BLE-RX] Buffer overflow! Discarding.");
            bleRxBuffer = "";
        }
        bleRxBuffer += chunk;

        String trimmed = bleRxBuffer;
        trimmed.trim();

        if (trimmed == "TRIGGER:DANGER" || trimmed == "CMD:CAPTURE_SNAPSHOT") {
            Serial.println("[BLE] Snapshot trigger received over BLE! Capturing...");
            bleRxBuffer = "";
            captureIncidentSnapshot();
            return;
        }

        if (trimmed.startsWith("{")) {
            if (isCompleteJsonObject(trimmed)) {
                String fullJson = trimmed;
                bleRxBuffer = "";

                // 1. Device Info Query
                if (fullJson.indexOf("get_device_info") >= 0) {
                    String infoMsg = "{\"type\":\"device_info\",\"device_id\":\"" + pin_device_id + 
                                     "\",\"mac\":\"" + pin_mac_address + 
                                     "\",\"fw_version\":\"2.1\",\"paired\":" + (isDevicePaired ? "true" : "false") + "}";
                    Serial.printf("[BLE-PROV] Replying with device info: %s\n", infoMsg.c_str());
                    sendBleText(infoMsg);
                    return;
                }

                // 2. Wi-Fi & Pairing Provisioning Command: Enqueue to dedicated worker task!
                if (fullJson.indexOf("\"provision\"") >= 0) {
                    // Send immediate non-blocking ACK
                    sendBleText("{\"type\":\"provision_received\"}");
                    Serial.println("[BLE-PROV] Provisioning command received and acknowledged to phone.");

                    if (isBleProvisioningActive) {
                        Serial.println("[BLE-PROV] Provisioning already in progress. Rejecting concurrent attempt.");
                        sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_BUSY\",\"message\":\"Provisioning task is already in progress\"}");
                        return;
                    }

                    String newSsid = extractJsonField(fullJson, "ssid");
                    String newPass = extractJsonField(fullJson, "password");
                    String pCode = extractJsonField(fullJson, "pairing_code");
                    String srvUrl = extractJsonField(fullJson, "server_url");

                    if (newSsid.length() == 0 || pCode.length() == 0) {
                        sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_INVALID_PARAMS\",\"message\":\"SSID and pairing_code required\"}");
                        return;
                    }

                    BleProvisionRequest req;
                    memset(&req, 0, sizeof(req));
                    strncpy(req.ssid, newSsid.c_str(), sizeof(req.ssid) - 1);
                    strncpy(req.password, newPass.c_str(), sizeof(req.password) - 1);
                    strncpy(req.pairing_code, pCode.c_str(), sizeof(req.pairing_code) - 1);
                    strncpy(req.server_url, srvUrl.c_str(), sizeof(req.server_url) - 1);

                    if (bleProvisionQueue != nullptr) {
                        if (xQueueSend(bleProvisionQueue, &req, 0) == pdTRUE) {
                            Serial.println("[BLE-PROV] Provisioning parameters enqueued to worker task.");
                        } else {
                            sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_QUEUE_FULL\",\"message\":\"Could not enqueue provisioning request\"}");
                        }
                    } else {
                        sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_INTERNAL\",\"message\":\"Provisioning queue not initialized\"}");
                    }
                    return;
                }
            }
            // Incomplete JSON fragment, continue buffering...
        } else {
            if (trimmed.indexOf("TRIGGER:DANGER") >= 0 || trimmed.indexOf("CMD:CAPTURE_SNAPSHOT") >= 0) {
                captureIncidentSnapshot();
            }
            bleRxBuffer = "";
        }
    }
};

void bleProvisionWorkerTask(void* param) {
    BleProvisionRequest req;
    while (true) {
        if (xQueueReceive(bleProvisionQueue, &req, portMAX_DELAY) == pdTRUE) {
            isBleProvisioningActive = true;
            String newSsid = String(req.ssid);
            String newPass = String(req.password);
            String pCode = String(req.pairing_code);
            String srvUrl = String(req.server_url);

            Serial.printf("\n[BLE-PROV-TASK] Provisioning worker running for SSID: '%s'\n", newSsid.c_str());
            sendBleText("{\"type\":\"wifi_connecting\",\"ssid\":\"" + newSsid + "\"}");

            // Attempt STA connection
            WiFi.disconnect(true);
            vTaskDelay(pdMS_TO_TICKS(100));
            WiFi.mode(WIFI_STA);
            WiFi.begin(newSsid.c_str(), newPass.c_str());

            unsigned long tStart = millis();
            bool wifiOk = false;
            while (millis() - tStart < 12000) {
                vTaskDelay(pdMS_TO_TICKS(150));
                if (WiFi.status() == WL_CONNECTED) {
                    wifiOk = true;
                    break;
                }
            }

            if (!wifiOk) {
                Serial.println("[BLE-PROV-TASK] Wi-Fi connection timed out or credentials rejected!");
                sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_WIFI_AUTH\",\"message\":\"Could not connect to Wi-Fi network\"}");
                isBleProvisioningActive = false;
                continue;
            }

            String localIp = WiFi.localIP().toString();
            Serial.printf("[BLE-PROV-TASK] Wi-Fi connected! Smart Pin IP: %s\n", localIp.c_str());
            sendBleText("{\"type\":\"wifi_connected\",\"ip\":\"" + localIp + "\"}");
            sendBleText("{\"type\":\"registration_pending\"}");

            // Parse target host & port
            String targetHost = server_host;
            int targetPort = server_port;
            if (srvUrl.length() > 0) {
                String clean = srvUrl;
                if (clean.startsWith("http://")) clean = clean.substring(7);
                else if (clean.startsWith("https://")) clean = clean.substring(8);
                int colonIdx = clean.indexOf(":");
                int slashIdx = clean.indexOf("/");
                if (colonIdx > 0) {
                    targetHost = clean.substring(0, colonIdx);
                    int endPort = (slashIdx > colonIdx) ? slashIdx : clean.length();
                    targetPort = clean.substring(colonIdx + 1, endPort).toInt();
                    if (targetPort <= 0) targetPort = 8080;
                } else if (slashIdx > 0) {
                    targetHost = clean.substring(0, slashIdx);
                } else {
                    targetHost = clean;
                }
            }

            // Contact Central System POST /api/pin/register with HTTPS if remote
            HTTPClient http;
            bool isRemoteHttps = (srvUrl.startsWith("https://") || targetHost.indexOf("trycloudflare.com") >= 0 || targetPort == 443);
            if (isRemoteHttps && (targetPort == 8080 || targetPort <= 0)) {
                targetPort = 443;
            }
            String regUrl = (isRemoteHttps ? "https://" : "http://") + targetHost + (isRemoteHttps ? "" : (":" + String(targetPort))) + "/api/pin/register";
            Serial.printf("[BLE-PROV-TASK] Contacting registration endpoint: %s\n", regUrl.c_str());

            WiFiClientSecure secureClient;
            WiFiClient plainClient;
            if (isRemoteHttps) {
                secureClient.setCACert(root_ca_isrg_x1);
                http.begin(secureClient, regUrl);
            } else {
                http.begin(plainClient, regUrl);
            }
            http.addHeader("Content-Type", "application/json");
            http.setTimeout(10000);

            String regPayload = "{\"device_id\":\"" + pin_device_id + "\",\"pairing_code\":\"" + pCode + "\",\"firmware_version\":\"2.1\",\"mac\":\"" + pin_mac_address + "\"}";
            int httpCode = http.POST(regPayload);
            String httpResp = http.getString();
            http.end();

            Serial.printf("[BLE-PROV-TASK] Central System HTTP response code: %d\n", httpCode);

            if (httpCode != 200 || httpResp.indexOf("\"status\":\"success\"") < 0) {
                String errMsg = "Pairing code rejected or invalid";
                if (httpResp.indexOf("\"message\":\"") >= 0) {
                    errMsg = extractJsonField(httpResp, "message");
                }
                sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_PAIRING_CODE\",\"message\":\"" + errMsg + "\"}");
                isBleProvisioningActive = false;
                continue;
            }

            String token = extractJsonField(httpResp, "device_token");
            if (token.length() < 16) {
                sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_INVALID_TOKEN\",\"message\":\"Central System returned invalid token\"}");
                isBleProvisioningActive = false;
                continue;
            }

            // Save persistent credentials to NVS Preferences
            prefs.putString("ssid", newSsid);
            prefs.putString("pass", newPass);
            prefs.putString("server", targetHost);
            prefs.putInt("port", targetPort);
            prefs.putString("token", token);

            // Verify persistence by reading back
            String readBackToken = prefs.getString("token", "");
            if (readBackToken != token || readBackToken.length() == 0) {
                Serial.println("[BLE-PROV-TASK] NVS save verification failed!");
                sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_NVS_FAILED\",\"message\":\"Failed to persist device token to flash memory\"}");
                isBleProvisioningActive = false;
                continue;
            }

            wifi_ssid = newSsid;
            wifi_pass = newPass;
            server_host = targetHost;
            server_port = targetPort;
            pin_device_token = token;
            cachedServerIP = targetHost;
            Serial.println("[BLE-PROV-TASK] Token and Wi-Fi credentials persisted to NVS.");

            // Connect authenticated WebSocket to Central System
            Serial.println("[BLE-PROV-TASK] Connecting to Central System WebSocket for authentication...");
            WiFi.setSleep(false);
            if (ws.isConnected()) {
                ws.disconnect();
                vTaskDelay(pdMS_TO_TICKS(200));
            }
            connectWebSocket(server_host, server_port);

            // Wait up to 10 seconds for WebSocket connection & authentication confirmation
            unsigned long authWaitStart = millis();
            bool authConfirmed = false;
            while (millis() - authWaitStart < 10000) {
                vTaskDelay(pdMS_TO_TICKS(150));
                if (isDevicePaired && ws.isConnected()) {
                    authConfirmed = true;
                    break;
                }
            }

            if (authConfirmed) {
                Serial.println("[BLE-PROV-TASK] Smart Pin successfully authenticated over WebSocket!");
                sendBleText("{\"type\":\"pairing_success\",\"device_id\":\"" + pin_device_id + "\"}");
            } else {
                Serial.println("[BLE-PROV-TASK] WebSocket authentication did not confirm within timeout.");
                if (ws.isConnected()) {
                    isDevicePaired = true;
                    sendBleText("{\"type\":\"pairing_success\",\"device_id\":\"" + pin_device_id + "\"}");
                } else {
                    sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_WS_AUTH\",\"message\":\"WebSocket authentication failed or timed out\"}");
                }
            }

            isBleProvisioningActive = false;
        }
    }
}

void setupBLE() {
    Serial.printf("[BLE] Starting Nordic UART BLE Service ('%s')...\n", ble_device_name.c_str());
    BLEDevice::init(ble_device_name.c_str());
    BLEDevice::setMTU(512);

    pBleServer = BLEDevice::createServer();
    pBleServer->setCallbacks(new BleServerCallbacks());

    BLEService *pService = pBleServer->createService(BLE_SERVICE_UUID);

    pTxCharacteristic = pService->createCharacteristic(
                          BLE_CHARACTERISTIC_UUID_TX,
                          BLECharacteristic::PROPERTY_NOTIFY
                        );
    pTxCharacteristic->addDescriptor(new BLE2902());

    pService->start();

    BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
    pAdvertising->addServiceUUID(BLE_SERVICE_UUID);

}

bool sendBleChunk(const uint8_t* data, size_t len) {
    if (!pTxCharacteristic || !bleClientConnected) return false;
    size_t sent = 0;
    while (sent < len && bleClientConnected) {
        size_t subChunk = min((size_t)500, len - sent);
        pTxCharacteristic->setValue((uint8_t*)&data[sent], subChunk);
        pTxCharacteristic->notify();
        sent += subChunk;
        delay(3);
    }
    return (sent == len);
}

// --- ACOUSTIC ENHANCEMENT FILTERS ---

// 1. 2nd-Order Butterworth High-Pass Filter (100Hz cutoff at 16kHz)
// Strips sub-audible fan rumble & physical handling noise (<100Hz) while keeping 100% of vocal fundamentals (100Hz-3400Hz)
inline int16_t applyHighPass(int16_t input) {
    static float x1 = 0.0f, x2 = 0.0f, y1 = 0.0f, y2 = 0.0f;
    float x0 = (float)input;
    float y0 = 0.972614f * x0 - 1.945228f * x1 + 0.972614f * x2 + 1.944478f * y1 - 0.945978f * y2;
    x2 = x1; x1 = x0;
    y2 = y1; y1 = y0;
    if (y0 > 32767.0f) y0 = 32767.0f;
    else if (y0 < -32768.0f) y0 = -32768.0f;
    return (int16_t)y0;
}

// 2. Soft Limiter (Prevents square-wave distortion on loud speech for Whisper)
inline int16_t applySoftLimiter(int32_t sample) {
    const int32_t limit = 26000;
    if (sample > limit) {
        int32_t val = limit + ((sample - limit) >> 2);
        return (int16_t)(val > 32767 ? 32767 : val);
    } else if (sample < -limit) {
        int32_t val = -limit + ((sample + limit) >> 2);
        return (int16_t)(val < -32768 ? -32768 : val);
    }
    return (int16_t)sample;
}

// --- BATTERY TELEMETRY (XIAO ESP32S3 Sense) ---
// Reads physical battery voltage from ADC (GPIO 1 / A0) with 1:2 divider,
// or calculates realistic Coulomb discharge based on 200mAh LiPo operational runtime.
#define BATTERY_PIN 1

static unsigned long totalStreamTimeMs = 0;
static unsigned long streamStartMs = 0;

int readBatteryMv() {
    analogReadResolution(12);
    analogSetPinAttenuation(BATTERY_PIN, ADC_11db);
    uint32_t rawSum = 0;
    for (int i = 0; i < 8; i++) {
        rawSum += analogRead(BATTERY_PIN);
        delayMicroseconds(50);
    }
    uint32_t raw = rawSum / 8;
    // With 11dB attenuation, range is ~0-3100mV.
    // If a 1:2 divider (e.g. 2x 100k resistors) is wired to GPIO 1:
    int mv = (int)((raw / 4095.0f) * 3100.0f * 2.0f);
    return mv;
}

int calculateBatteryPct() {
    int mv = readBatteryMv();
    // If a valid physical LiPo voltage (3.0V - 4.35V) is measured via hardware divider:
    if (mv >= 3000 && mv <= 4350) {
        int pct = (int)(((float)(mv - 3300) / (4200 - 3300)) * 100.0f);
        return constrain(pct, 0, 100);
    }

    // Dynamic Coulomb discharge model for 200mAh LiPo:
    // Standby: ~35mA (160MHz CPU + Modem Sleep + OV2640 in powersave) -> ~5.7h total
    // Active Voice Streaming: ~115mA -> ~1.7h total
    unsigned long now = millis();
    unsigned long activeStreamMs = totalStreamTimeMs;
    if (isStreaming && streamStartMs > 0) {
        activeStreamMs += (now - streamStartMs);
    }
    unsigned long standbyMs = (now > activeStreamMs) ? (now - activeStreamMs) : 0;

    float consumedMah = ((float)standbyMs / 3600000.0f) * 35.0f + ((float)activeStreamMs / 3600000.0f) * 115.0f;
    float remainingMah = 200.0f - consumedMah;
    int pct = (int)((remainingMah / 200.0f) * 100.0f);
    return constrain(pct, 0, 100);
}

// Non-blocking snapshot engine:
// If camera hardware freezes or drops DMA, Core 0 audio streaming will NEVER hang!
static volatile bool snapshotInProgress = false;
static unsigned long snapshotStartedAt = 0;
static camera_fb_t* volatile readySnapshotFb = nullptr;

void snapshotWorkerTask(void* param) {
    // Drop the pre-buffered frame in FIFO to ensure the snapshot is fresh
    camera_fb_t * staleFb = esp_camera_fb_get();
    if (staleFb) {
        esp_camera_fb_return(staleFb);
    }

    // Capture fresh evidence snapshot from OV3660 sensor
    camera_fb_t * fb = esp_camera_fb_get();
    if (fb) {
        readySnapshotFb = fb;
        Serial.printf("[CAM] OV3660 snapshot captured (%d bytes, %dx%d)\n", fb->len, fb->width, fb->height);
    } else {
        Serial.println("[CAM] esp_camera_fb_get() returned null (capture failed).");
        snapshotInProgress = false;
    }
    vTaskDelete(NULL);
}

void captureIncidentSnapshot() {
    if (!cameraReady) {
        Serial.println("[CAM] Snapshot skipped: OV3660 camera not ready.");
        return;
    }
    if (snapshotInProgress) {
        if (millis() - snapshotStartedAt > 3500) {
            Serial.println("[CAM] Warning: Previous snapshot task timed out. Resetting state.");
            snapshotInProgress = false;
        } else {
            Serial.println("[CAM] Snapshot already in progress, skipping.");
            return;
        }
    }

    snapshotInProgress = true;
    snapshotStartedAt = millis();
    Serial.println("[CAM] Firing OV3660 snapshot task on Core 0...");
    BaseType_t res = xTaskCreatePinnedToCore(
        snapshotWorkerTask,
        "camSnapTask",
        4096,
        NULL,
        1,
        NULL,
        0
    );
    if (res != pdPASS) {
        snapshotInProgress = false;
        Serial.println("[CAM] Failed to create camSnapTask.");
    }
}

void onWebSocketEvent(WStype_t type, uint8_t* payload, size_t length) {
    switch (type) {
        case WStype_CONNECTED:
            hasActiveStream = false; // Standby connection ready! Audio streaming begins only when voice is detected
            hasPendingWsConnect = false;
            isStreaming = false; // Always clear stale streaming state on connect
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            needImmediateHeartbeat = true; // Signal main loop to transmit initial PING cleanly once handshake finishes
            Serial.println("[WS] Connected to Checkpoint in Standby (VAD Armed)");
            break;
        case WStype_TEXT:
            if (length > 0) {
                String msg = String((char*)payload);
                if (msg == "TRIGGER:DANGER" || msg == "CMD:CAPTURE_SNAPSHOT") {
                    Serial.println("[CAM] Speech evaluated as DANGER! Capturing camera snapshot...");
                    captureIncidentSnapshot();
                } else if (msg.startsWith("CONFIG:BUFFER:")) {
                    int sec = msg.substring(14).toInt();
                    if (sec > 0) {
                        SILENCE_TIMEOUT = sec * 1000;
                        Serial.printf("[CFG] Silence Timeout updated to: %lu ms\n", SILENCE_TIMEOUT);
                    }
                } else if (msg.startsWith("CONFIG:WIFI:")) {
                    // Format: CONFIG:WIFI:<ssid>:<pass>:<server_ip>:<port>
                    int p1 = msg.indexOf(':', 12);
                    int p2 = msg.indexOf(':', p1 + 1);
                    int p3 = msg.indexOf(':', p2 + 1);
                    if (p1 > 0 && p2 > 0) {
                        String n_ssid = msg.substring(12, p1);
                        String n_pass = msg.substring(p1 + 1, p2);
                        String n_server = p3 > 0 ? msg.substring(p2 + 1, p3) : msg.substring(p2 + 1);
                        int n_port = p3 > 0 ? msg.substring(p3 + 1).toInt() : 8080;
                        if (n_port <= 0) n_port = 8080;

                        Serial.printf("\n[CFG] >>> RECEIVED NEW WI-FI CHECKPOINT OVER WEBSOCKET! <<<\n");
                        Serial.printf("[CFG] New SSID: '%s' | Server: '%s:%d'\n", n_ssid.c_str(), n_server.c_str(), n_port);

                        // Save current network as backup/secondary checkpoint before switching
                        if (wifi_ssid.length() > 0 && wifi_ssid != n_ssid) {
                            prefs.putString("ssid2", wifi_ssid);
                            prefs.putString("pass2", wifi_pass);
                            wifi_ssid2 = wifi_ssid;
                            wifi_pass2 = wifi_pass;
                        }

                        prefs.putString("ssid", n_ssid);
                        prefs.putString("pass", n_pass);
                        prefs.putString("server", n_server);
                        prefs.putInt("port", n_port);

                        wifi_ssid = n_ssid;
                        wifi_pass = n_pass;
                        server_host = n_server;
                        server_port = n_port;
                        cachedServerIP = n_server;

                        ws.sendTXT("OK:CONFIG_WIFI_SAVED");
                        delay(400);
                        ws.disconnect();
                        delay(200);
                        Serial.println("[SYS] Rebooting to connect to updated checkpoint...");
                        ESP.restart();
                    }
                } else if (msg == "CMD:FORGET_WIFI" || msg == "CMD:RESET_WIFI") {
                    Serial.println("\n[SYS] >>> RECEIVED 'CMD:FORGET_WIFI' FROM DASHBOARD! <<<");
                    Serial.println("[SYS] Erasing Wi-Fi credentials from NVS flash...");
                    prefs.remove("ssid");
                    prefs.remove("pass");
                    prefs.remove("ssid2");
                    prefs.remove("pass2");
                    prefs.remove("token");
                    wifi_ssid = "";
                    wifi_pass = "";
                    wifi_ssid2 = "";
                    wifi_pass2 = "";
                    pin_device_token = "";
                    isDevicePaired = false;
                    ws.disconnect();
                    delay(500);
                    Serial.println("[SYS] Rebooting into Setup Portal...");
                    ESP.restart();
                } else if (msg.indexOf("\"type\":\"auth_ok\"") >= 0) {
                    isDevicePaired = true;
                    Serial.println("[WS-AUTH] WebSocket Authentication Confirmed by Central System!");
                    sendBleText("{\"type\":\"pairing_success\",\"device_id\":\"" + pin_device_id + "\"}");
                } else if (msg.indexOf("\"type\":\"auth_error\"") >= 0) {
                    Serial.println("[WS-AUTH] WebSocket Authentication Rejected by Central System!");
                    sendBleText("{\"type\":\"provision_error\",\"code\":\"ERR_WS_AUTH\",\"message\":\"Central System rejected token\"}");
                } else if (msg.indexOf("\"type\":\"pairing_success\"") >= 0) {
                    String token = extractJsonField(msg, "device_token");
                    if (token.length() > 0) {
                        prefs.putString("token", token);
                        pin_device_token = token;
                        isDevicePaired = true;
                        Serial.println("[WS-AUTH] Smart Pin token received and stored from Central System!");
                    }
                } else if (msg.indexOf("\"type\":\"auth_challenge\"") >= 0) {
                    if (pin_device_token.length() > 0) {
                        ws.sendTXT("{\"type\":\"auth\",\"device_id\":\"" + pin_device_id + "\",\"token\":\"" + pin_device_token + "\"}");
                    }
                }
            }
            break;
        case WStype_DISCONNECTED:
            hasActiveStream = false;
            hasPendingWsConnect = false;
            isStreaming = false; // Always clear streaming state on disconnect
            isUploadingArchive = false;
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            if (uploadFile) uploadFile.close();
            Serial.println("[WS] Disconnected");
            break;
        default:
            break;
    }
}

void setupMicrophone() {
    I2S.setPinsPdmRx(42, 41);
    if (!I2S.begin(I2S_MODE_PDM_RX, SAMPLE_RATE, I2S_DATA_BIT_WIDTH_16BIT, I2S_SLOT_MODE_MONO)) {
        Serial.println("[ERR] I2S Init Failed!");
        while(1) delay(100);
    }
    speex_st = speex_preprocess_state_init(FRAME_SIZE, SAMPLE_RATE);
    int vadEnabled = 1;
    int probStart = 80;    // 80: High confidence speech onset trigger
    int probContinue = 60; // 60: Drops cleanly when voice stops
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_VAD, &vadEnabled);
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_START, &probStart);
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_PROB_CONTINUE, &probContinue);

    // Denoising is handled by Microsoft NSNet2 on the GPU/server.
    // Keeping Speex denoise OFF on the ESP32 frees 80% CPU on Core 1 and eliminates I2S DMA dropouts!
    int denoise = 0;
    speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_SET_DENOISE, &denoise);
}

void discoverCheckpoint() {
    if (hasActiveStream || hasPendingWsConnect || ws.isConnected() || wifiConnectInProgress) return;

    if (!isDevicePaired || pin_device_token.length() == 0) {
        static unsigned long lastUnpairedLog = 0;
        if (millis() - lastUnpairedLog > 10000) {
            lastUnpairedLog = millis();
            Serial.printf("[WS] Pin '%s' connected to Wi-Fi but unpaired (no token). Awaiting pairing via Mobile BLE or Setup Portal.\n", pin_device_id.c_str());
        }
        return;
    }

    if (WiFi.status() != WL_CONNECTED) {
        wifiConnectInProgress = true;
        wifiConnectStartedAt = millis();
        WiFi.mode(WIFI_STA);
        WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        Serial.printf("[WIFI] Connecting to %s...\n", wifi_ssid.c_str());
        return;
    }

    if (cachedServerIP != "") {
        connectWebSocket(cachedServerIP, server_port);
        return;
    }

    udp.begin(8888);
    udp.beginPacket(IPAddress(255, 255, 255, 255), 8888);
    udp.print("SECURELINK_PIN_V1");
    udp.endPacket();

    // Give server up to 800ms to respond over Wi-Fi (50ms was too fast)
    unsigned long waitStart = millis();
    while (millis() - waitStart < 800) {
        if (udp.parsePacket()) {
            cachedServerIP = udp.remoteIP().toString();
            Serial.printf("[UDP] Found Checkpoint Server at: %s\n", cachedServerIP.c_str());
            connectWebSocket(cachedServerIP, server_port);
            break;
        }
        delay(20);
    }
}

// Ensure space exists before local recording: deletes oldest timestamped incident
void enforceFreeSpace() {
    if (!fsReady) return;
    
    while ((LittleFS.totalBytes() - LittleFS.usedBytes()) < 400000) {
        File root = LittleFS.open("/");
        File file = root.openNextFile();
        if (!file) break;
        
        String oldestFile = "";
        unsigned long oldestTime = 0xFFFFFFFF;

        while (file) {
            String name = String(file.name());
            if (name.startsWith("inc_") && name.endsWith(".pcm")) {
                int start = name.indexOf('_') + 1;
                int end = name.indexOf('.');
                unsigned long t = name.substring(start, end).toInt();
                if (t < oldestTime) {
                    oldestTime = t;
                    oldestFile = "/" + name;
                }
            }
            file = root.openNextFile();
        }
        root.close();

        if (oldestFile != "") {
            LittleFS.remove(oldestFile);
            Serial.println("[FS] Low space. Removed oldest file: " + oldestFile);
        } else {
            break;
        }
    }
}

void processArchiveUpload() {
    // Only upload offline recordings when connected to WebSocket, idle (not actively streaming live speech), and flash is ready
    if (!ws.isConnected() || isStreaming || isOfflineRecording || !fsReady) return;

    if (!isUploadingArchive) {
        File root = LittleFS.open("/");
        File file = root.openNextFile();
        while (file) {
            String name = String(file.name());
            if (name.endsWith(".pcm")) {
                uploadFilePath = "/" + name;
                uploadFile = LittleFS.open(uploadFilePath, FILE_READ);
                isUploadingArchive = true;
                ws.sendTXT("ARCHIVE_START:" + uploadFilePath);
                Serial.printf("[INC] Syncing stored incident: %s\n", uploadFilePath.c_str());
                file.close();
                root.close();
                return;
            }
            file = root.openNextFile();
        }
        root.close();
        return; 
    }

    static unsigned long lastArchiveChunkMs = 0;
    if (millis() - lastArchiveChunkMs < 40) return; // 40ms timer pace = 51.2 KB/s (safely below server 80 KB/s rate limit)

    if (uploadFile.available()) {
        uint8_t buf[2048];
        size_t bytesRead = uploadFile.read(buf, sizeof(buf));
        if (bytesRead > 0) {
            ws.sendBIN(buf, bytesRead);
            lastArchiveChunkMs = millis();
        }
    } else {
        uploadFile.close();
        LittleFS.remove(uploadFilePath);
        ws.sendTXT("ARCHIVE_END:" + uploadFilePath);
        Serial.printf("[INC] Sync complete. Deleted: %s\n", uploadFilePath.c_str());
        isUploadingArchive = false;
        lastArchiveChunkMs = 0;
    }
}

void handleSerialCommands() {
    if (Serial.available()) {
        String cmd = Serial.readStringUntil('\n');
        cmd.trim();
        if (cmd.startsWith("SET_WIFI:") || cmd.startsWith("CONFIG_WIFI:")) {
            // Format: SET_WIFI:<ssid>:<pass>:<server_ip>:<port>[:<pairing_code>]
            int p1 = cmd.indexOf(':');
            int p2 = cmd.indexOf(':', p1 + 1);
            int p3 = cmd.indexOf(':', p2 + 1);
            int p4 = cmd.indexOf(':', p3 + 1);
            int p5 = (p4 > 0) ? cmd.indexOf(':', p4 + 1) : -1;
            if (p1 > 0 && p2 > 0) {
                String n_ssid = cmd.substring(p1 + 1, p2);
                String n_pass = cmd.substring(p2 + 1, p3 > 0 ? p3 : cmd.length());
                String n_server = p3 > 0 ? cmd.substring(p3 + 1, p4 > 0 ? p4 : cmd.length()) : server_host;
                int n_port = p4 > 0 ? cmd.substring(p4 + 1, p5 > 0 ? p5 : cmd.length()).toInt() : server_port;
                if (n_port <= 0) n_port = 8080;
                String n_pcode = p5 > 0 ? cmd.substring(p5 + 1) : "";
                n_pcode.trim();
                if (n_pcode.length() == 6) {
                    prefs.putString("pcode_pending", n_pcode);
                }

                // If primary Wi-Fi is changing, save previous as secondary roaming backup
                if (wifi_ssid.length() > 0 && wifi_ssid != n_ssid) {
                    prefs.putString("ssid2", wifi_ssid);
                    prefs.putString("pass2", wifi_pass);
                    wifi_ssid2 = wifi_ssid;
                    wifi_pass2 = wifi_pass;
                }

                prefs.putString("ssid", n_ssid);
                prefs.putString("pass", n_pass);
                prefs.putString("server", n_server);
                prefs.putInt("port", n_port);

                wifi_ssid = n_ssid;
                wifi_pass = n_pass;
                server_host = n_server;
                server_port = n_port;
                cachedServerIP = n_server;

                Serial.printf("[SYS] OK:CONFIG_SAVED - Primary:'%s' Backup:'%s' Host:'%s:%d'\n", wifi_ssid.c_str(), wifi_ssid2.c_str(), server_host.c_str(), server_port);
                delay(300);
                ESP.restart();
            } else {
                Serial.println("[ERR] Invalid SET_WIFI format. Use: SET_WIFI:<ssid>:<pass>:<server_ip>:<port>");
            }
        } else if (cmd.equalsIgnoreCase("format")) {
            Serial.println("[FS] Formatting LittleFS. Please wait...");
            if (LittleFS.format()) {
                Serial.println("[FS] Format Complete! Storage is empty.");
            } else {
                Serial.println("[ERR] Format Failed.");
            }
        } else if (cmd.equalsIgnoreCase("setup") || cmd.equalsIgnoreCase("portal") || cmd.equalsIgnoreCase("wifi") || cmd.equalsIgnoreCase("config")) {
            Serial.println("[SYS] Setup command received. Starting Web Portal...");
            startConfigPortal();
        } else if (cmd.equalsIgnoreCase("status")) {
            Serial.printf("[STATUS] SSID:%s IP:%s WS:%d VAD:%d CAM:%d\n", wifi_ssid.c_str(), WiFi.localIP().toString().c_str(), (int)hasActiveStream, (int)lastVadState, (int)cameraReady);
        } else if (cmd.equalsIgnoreCase("snap") || cmd.equalsIgnoreCase("capture") || cmd.equalsIgnoreCase("danger")) {
            Serial.println("[CAM] Manual snapshot triggered via Serial command!");
            captureIncidentSnapshot();
        }
    }
}

// =========================================================================
// CORE 1: DEDICATED REAL-TIME AUDIO ENGINE
// Continuously samples I2S, executes DSP + Speex VAD, and appends to PSRAM
// ring buffer. Completely isolated from Wi-Fi/TCP jitter on Core 0!
// =========================================================================
void audioRecordTask(void* pvParameters) {
    Serial.println("[AUDIO] Core 1 Audio Engine task running at Priority 5.");
    while (true) {
        size_t bytesRead = I2S.readBytes((char*)audioBuffer, sizeof(audioBuffer));
        if (bytesRead != sizeof(audioBuffer)) {
            vTaskDelay(pdMS_TO_TICKS(2));
            continue;
        }

        // 1. Apply High-Pass Filter & Peak Tracking
        int32_t currentPeak = 0;
        for (int i = 0; i < FRAME_SIZE; i++) {
            int16_t filtered = applyHighPass(audioBuffer[i]);
            audioBuffer[i] = filtered; // Clean audio for transmission
            vadBuffer[i] = filtered;
            if (abs(filtered) > currentPeak) currentPeak = abs(filtered);
        }

        // 2. Speex VAD
        speex_preprocess_run(speex_st, vadBuffer);
        int isSpeech = 0;
        speex_preprocess_ctl(speex_st, SPEEX_PREPROCESS_GET_VAD, &isSpeech);

        // Speech detection criteria:
        // 1) Speex VAD confirms vocal fold harmonics (isSpeech != 0)
        // 2) Amplitude rises significantly above ambient baseline (+50)
        // 3) Absolute peak >= 220 (ignores mic idle floor 60-160 and fan hum)
        // 4) Amplitude fallback bypass (currentPeak >= 450): prevents Speex zero-noise silence freezes
        bool speechDetected = ((isSpeech != 0) && (currentPeak > ambientFloor + 50) && (currentPeak >= 220)) || (currentPeak >= 450);

        // VAD Hangover Engine: 400ms (20 frames) hangover bridges natural breath pauses & consonants
        if (speechDetected) {
            if (speechStreak < 4) speechStreak++;
            if (speechStreak >= SPEECH_STREAK_REQUIRED) {
                hangoverFrames = 20;
            }
        } else {
            if (speechStreak > 0) speechStreak--;
            if (hangoverFrames > 0) hangoverFrames--;
        }

        speechArmed = (speechStreak >= SPEECH_STREAK_REQUIRED) || (hangoverFrames > 0);

        // Dynamic ambient floor tracking: adaptively track background when speech is inactive
        if (!speechArmed && isSpeech == 0) {
            ambientFloor = (ambientFloor * 31 + currentPeak) / 32;
            if (ambientFloor < 20) ambientFloor = 20;
            if (ambientFloor > 200) ambientFloor = 200;
        }

        if (speechArmed != lastVadState) {
            Serial.println(speechArmed ? "[VAD] VOICE DETECTED" : "[VAD] SILENCE");
            lastVadState = speechArmed;
        }

        // 3. Format clean 16-bit PCM for transmission
        // Passing full natural dynamic range directly preserves vocal warmth and eliminates raspy distortion
        uint8_t outBuf[FRAME_SIZE * sizeof(int16_t)];
        for (int i = 0; i < FRAME_SIZE; i++) {
            int16_t sample = audioBuffer[i];
            outBuf[i * 2] = sample & 0xFF;
            outBuf[i * 2 + 1] = (sample >> 8) & 0xFF;
        }

        // 4. Lock-free append to PSRAM Ring Buffer
        if (audioRingBuffer != nullptr) {
            for (size_t i = 0; i < sizeof(outBuf); i++) {
                audioRingBuffer[(ringHead + i) % RING_BUFFER_SIZE] = outBuf[i];
            }
            ringHead = (ringHead + sizeof(outBuf)) % RING_BUFFER_SIZE;

            // When NOT streaming, keep ringTail floating exactly PRE_ROLL_BYTES behind ringHead
            if (!isStreaming) {
                ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
            } else {
                // Buffer Overrun Guard: If ringHead gets dangerously close to ringTail (e.g. Wi-Fi stalled),
                // advance ringTail to drop oldest unread data cleanly and protect live speech continuity.
                size_t used = (ringHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;
                if (used >= RING_BUFFER_SIZE - (BATCH_SIZE * 2)) {
                    ringTail = (ringTail + BATCH_SIZE) % RING_BUFFER_SIZE;
                }
            }
        }

        taskYIELD(); // Yield FreeRTOS slice to avoid starving RTOS scheduler and watchdogs
    }
}

void setup() {
    // 1. BALANCED 160MHz CLOCK FOR OPTIMAL BATTERY LIFE & LOW THERMAL FOOTPRINT
    setCpuFrequencyMhz(160);

    Serial.begin(115200);
    delay(500);
    Serial.printf("[SYS] CPU Clock: %d MHz\n", getCpuFrequencyMhz());

    pinMode(0, INPUT_PULLUP); // XIAO ESP32S3 Boot button
    loadPreferences();
    initDeviceIdentity(); // Compute persistent hardware ID & BLE device name

    // 2. CRITICAL FIX: INITIALIZE BLE FIRST!
    // BLE is active immediately upon boot so phone can discover and pair without buttons or cables
    setupBLE();

    // If Boot button is manually held at power-on, enter fallback captive portal
    if (digitalRead(0) == LOW) {
        Serial.println("[SYS] Boot button held at power-on! Starting manual setup portal...");
        startConfigPortal();
    }

    bootStartMs = millis();

    // 3. INITIALIZE PSRAM RING BUFFER
    if (psramInit()) {
        audioRingBuffer = (uint8_t*)ps_malloc(RING_BUFFER_SIZE);
        Serial.printf("[SYS] 8MB PSRAM found! Ring Buffer allocated in PSRAM: %u bytes\n", (unsigned int)RING_BUFFER_SIZE);
    } else {
        audioRingBuffer = (uint8_t*)malloc(RING_BUFFER_SIZE);
        Serial.printf("[WARN] PSRAM not found. Allocated %u bytes in SRAM\n", (unsigned int)RING_BUFFER_SIZE);
    }

    if (!audioRingBuffer) {
        Serial.println("[ERR] Ring buffer allocation failed!");
        while(1) delay(100);
    }

    setupMicrophone();
    setupCamera();

    // Launch Dedicated Audio Engine on Core 1 (Core 0 handles Wi-Fi, WebSocket, LittleFS, and System)
    xTaskCreatePinnedToCore(
        audioRecordTask,
        "AudioEngine",
        8192,
        NULL,
        5,
        &audioTaskHandle,
        1
    );
    
    if (LittleFS.begin(true)) {
        fsReady = true;
        Serial.printf("[FS] Flash mounted. Total: %u bytes, Used: %u bytes\n", 
            LittleFS.totalBytes(), LittleFS.usedBytes());
    } else {
        Serial.println("[ERR] LittleFS Mount Failed");
    }

    // 4. BOOT WI-FI CONNECTION CHECK & ROAMING FALLBACK
    bool connected = false;
    if (wifi_ssid.length() > 0) {
        Serial.printf("\n[WIFI] Checking primary checkpoint '%s'...\n", wifi_ssid.c_str());
        WiFi.mode(WIFI_STA);
        WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        unsigned long t0 = millis();
        while (WiFi.status() != WL_CONNECTED && millis() - t0 < 10000) {
            delay(100);
            Serial.print(".");
            handleSerialCommands();
        }
        Serial.println();
        if (WiFi.status() == WL_CONNECTED) {
            connected = true;
            Serial.printf("[WIFI] Connected to Primary! Smart Pin IP: %s\n", WiFi.localIP().toString().c_str());
        }
    }

    if (!connected && wifi_ssid2.length() > 0) {
        Serial.printf("\n[WIFI] Trying roaming backup checkpoint '%s'...\n", wifi_ssid2.c_str());
        WiFi.disconnect(true);
        delay(100);
        WiFi.begin(wifi_ssid2.c_str(), wifi_pass2.c_str());
        unsigned long t0 = millis();
        while (WiFi.status() != WL_CONNECTED && millis() - t0 < 8000) {
            delay(100);
            Serial.print(".");
            handleSerialCommands();
        }
        Serial.println();
        if (WiFi.status() == WL_CONNECTED) {
            connected = true;
            Serial.printf("[WIFI] Connected to Backup Checkpoint! Smart Pin IP: %s\n", WiFi.localIP().toString().c_str());
        }
    }

    if (connected) {
        WiFi.setSleep(false); // Disable modem sleep: keeps TCP socket open and eliminates Windows WinError 121 timeouts
        if (!isDevicePaired && prefs.isKey("pcode_pending")) {
            String pendingCode = prefs.getString("pcode_pending", "");
            prefs.remove("pcode_pending");
            if (pendingCode.length() >= 6) {
                Serial.printf("[PROV] Found pending pairing code. Registering with Central System...\n");
                HTTPClient http;
                bool isRemoteHttps = (server_host.indexOf("trycloudflare.com") >= 0 || server_port == 443);
                String regUrl = (isRemoteHttps ? "https://" : "http://") + server_host + (isRemoteHttps ? "" : (":" + String(server_port))) + "/api/pin/register";
                WiFiClientSecure secClient;
                WiFiClient stdClient;
                if (isRemoteHttps) {
                    secClient.setCACert(root_ca_isrg_x1);
                    http.begin(secClient, regUrl);
                } else {
                    http.begin(stdClient, regUrl);
                }
                http.addHeader("Content-Type", "application/json");
                http.setTimeout(10000);
                String regPayload = "{\"device_id\":\"" + pin_device_id + "\",\"pairing_code\":\"" + pendingCode + "\",\"firmware_version\":\"2.1\",\"mac\":\"" + pin_mac_address + "\"}";
                int hCode = http.POST(regPayload);
                String hResp = http.getString();
                http.end();
                if (hCode == 200 && hResp.indexOf("\"status\":\"success\"") >= 0) {
                    String token = extractJsonField(hResp, "device_token");
                    if (token.length() > 0) {
                        prefs.putString("token", token);
                        String readBack = prefs.getString("token", "");
                        if (readBack == token && token.length() > 0) {
                            pin_device_token = token;
                            Serial.println("[PROV] Smart Pin token registered & verified in NVS. Connecting WebSocket to authenticate...");
                        } else {
                            Serial.println("[PROV] ERROR: Failed to persist device token to flash memory!");
                        }
                    }
                }
            }
        }
        discoverCheckpoint();
    } else if (wifi_ssid.length() == 0) {
        // BLE Onboarding is active: do not block in captive portal!
        Serial.printf("[WIFI] No Wi-Fi configured. Pin is in BLE Onboarding Mode ('%s').\n", ble_device_name.c_str());
        Serial.println("[WIFI] Ready for Mobile Companion BLE pairing!");
    } else {
        Serial.println("[WIFI] Initial boot connection pending. Proceeding to non-blocking background retry in loop()...");
    }
}

void loop() {
    handleSerialCommands();

    // Check for Boot button held for 2 seconds anytime to enter setup portal
    static unsigned long btnHoldStart = 0;
    if (digitalRead(0) == LOW) {
        if (btnHoldStart == 0) btnHoldStart = millis();
        else if (millis() - btnHoldStart > 2000) {
            btnHoldStart = 0;
            Serial.println("\n[SYS] Boot button held for 2s! Launching Setup Portal...");
            startConfigPortal();
        }
    } else {
        btnHoldStart = 0;
    }

    // Maintain BLE advertising if phone companion disconnected
    if (!bleClientConnected && oldBleClientConnected) {
        delay(300);
        pBleServer->startAdvertising();
        Serial.println("[BLE] Restarted advertising after companion disconnect.");
        oldBleClientConnected = bleClientConnected;
    }
    if (bleClientConnected && !oldBleClientConnected) {
        oldBleClientConnected = bleClientConnected;
    }

    // Check for completed non-blocking snapshot frame (Wi-Fi & BLE Multi-Channel Routing)
    if (readySnapshotFb != nullptr) {
        camera_fb_t* fb = readySnapshotFb;
        readySnapshotFb = nullptr;
        if (ws.isConnected()) {
            // Wi-Fi Route: WebSocket Text Header + Binary Buffer
            String header = "SNAP:" + String(fb->len);
            ws.sendTXT(header);
            ws.sendBIN(fb->buf, fb->len);
            Serial.printf("[CAM] Incident snapshot sent to server over Wi-Fi (%d bytes)\n", fb->len);
        } else if (bleClientConnected) {
            // BLE Route: Nordic UART Service sliced to MTU 512
            String header = "SNAP:" + String(fb->len);
            sendBleText(header);
            delay(20);
            sendBleChunk(fb->buf, fb->len);
            Serial.printf("[CAM] Incident snapshot sent to Mobile Companion over BLE (%d bytes)\n", fb->len);
        }
        esp_camera_fb_return(fb);
        snapshotInProgress = false;
    }

    static bool wifiWasDisconnected = false;
    if (WiFi.status() == WL_CONNECTED) {
        if (wifiWasDisconnected) {
            WiFi.setSleep(false); // Re-disable modem sleep after reconnect to eradicate WinError 121
            wifiWasDisconnected = false;
            Serial.println("[WIFI] Reconnected! Modem sleep disabled to prevent WinError 121.");
        }
        ws.loop();

        if (hasPendingWsConnect && ws.isConnected()) {
            hasPendingWsConnect = false;
        }
        if (hasPendingWsConnect && (millis() - wsConnectStartedAt > WS_CONNECT_TIMEOUT_MS)) {
            ws.disconnect();
            hasPendingWsConnect = false;
            cachedServerIP = server_host; 
        }
        // Auto-reconnect to checkpoint server on standby so streaming is instantaneous (only if device is paired)
        if (!ws.isConnected() && !hasPendingWsConnect && cachedServerIP != "" && !wifiConnectInProgress && isDevicePaired && pin_device_token.length() > 0) {
            static unsigned long lastWsRetry = 0;
            if (millis() - lastWsRetry > 3000) {
                lastWsRetry = millis();
                connectWebSocket(cachedServerIP, server_port);
            }
        }
    } else {
        wifiWasDisconnected = true;
        // Purge dead/zombie WebSocket state immediately if Wi-Fi drops
        if (ws.isConnected() || hasPendingWsConnect || hasActiveStream || isStreaming) {
            ws.disconnect();
            hasActiveStream = false;
            isStreaming = false;
            hasPendingWsConnect = false;
            Serial.println("[WS] Wi-Fi lost! Forced disconnect and reset all streaming flags.");
        }

        // Resilient background Wi-Fi auto-reconnection:
        // If Wi-Fi is temporarily lost, quietly retry every 5s. NEVER auto-lock into Setup Portal!
        static unsigned long lastWifiRetry = 0;
        if (millis() - lastWifiRetry > 5000) {
            lastWifiRetry = millis();
            Serial.printf("[WIFI] Reconnecting to '%s'...\n", wifi_ssid.c_str());
            WiFi.disconnect(true);
            delay(50);
            WiFi.mode(WIFI_STA);
            WiFi.begin(wifi_ssid.c_str(), wifi_pass.c_str());
        }
    }

    // --- AUDIO STREAMING & RECORDING PIPELINE (Core 0 Network Consumer) ---
    // TRIGGER STREAMING WHEN VOICE IS DETECTED (3-TIER ARCHITECTURE)
    if (speechArmed) {
        lastSpeechTime = millis();
        if (!isStreaming) {
            if (ws.isConnected()) {
                // Tier A: Direct Wi-Fi WebSocket
                hasActiveStream = true;
                isStreaming = true;
                WiFi.setSleep(false); // Active speech: wake Wi-Fi radio for zero-latency, reliable streaming
                streamStartMs = millis();
                Serial.println("[VAD] Voice triggered! Streaming via Wi-Fi WebSocket (Tier A).");
            } else if (bleClientConnected) {
                // Tier B: BLE Audio Bridge to Mobile Companion (Cellular/Enterprise Relay)
                hasActiveStream = false;
                isStreaming = true;
                streamStartMs = millis();
                Serial.println("[VAD] Voice triggered! Streaming via BLE Audio Bridge to Phone (Tier B).");
            } else {
                discoverCheckpoint();
                if (ws.isConnected()) {
                    hasActiveStream = true;
                    isStreaming = true;
                    WiFi.setSleep(false);
                    streamStartMs = millis();
                    Serial.println("[VAD] Voice triggered after checkpoint discovery! Streaming started.");
                } else if (!wifiConnectInProgress && fsReady) {
                    // Tier C: LittleFS Offline Blackbox
                    enforceFreeSpace();
                    isOfflineRecording = true;
                    isStreaming = true;
                    streamStartMs = millis();
                    currentIncidentPath = "/inc_" + String(millis()) + ".pcm";
                    offlineIncidentFile = LittleFS.open(currentIncidentPath, FILE_WRITE);
                    Serial.printf("[VAD] Wi-Fi & BLE unavailable. Recording locally to %s (Tier C).\n", currentIncidentPath.c_str());
                }
            }
        }
    }

    // ACTIVE STREAMING / LOCAL RECORDING ENGINE (3-TIER MULTI-CHANNEL)
    if (isStreaming) {
        size_t available = (ringHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;

        bool maxStreamReached = (streamStartMs > 0 && (millis() - streamStartMs > 8000));
        if ((millis() - lastSpeechTime > SILENCE_TIMEOUT) || maxStreamReached) {
            // Voice ended or max stream limit reached: drain remaining audio with verified delivery
            size_t targetHead = ringHead;
            size_t toDrain = (targetHead + RING_BUFFER_SIZE - ringTail) % RING_BUFFER_SIZE;
            int failedAttempts = 0;
            while (toDrain >= sizeof(int16_t) && failedAttempts < 5) {
                if (!ws.isConnected() && !bleClientConnected && !isOfflineRecording) break; // Abort drain immediately if no channel available
                size_t chunk = min((size_t)BATCH_SIZE, toDrain);
                uint8_t tempBuf[chunk];
                for (size_t i = 0; i < chunk; i++) {
                    tempBuf[i] = audioRingBuffer[(ringTail + i) % RING_BUFFER_SIZE];
                }

                bool sendOk = false;
                if (hasActiveStream && ws.isConnected()) {
                    // Tier A
                    sendOk = ws.sendBIN(tempBuf, chunk);
                } else if (bleClientConnected) {
                    // Tier B (BLE Notification sliced to MTU 512)
                    sendOk = sendBleChunk(tempBuf, chunk);
                } else if (isOfflineRecording && offlineIncidentFile) {
                    // Tier C (LittleFS Blackbox)
                    sendOk = (offlineIncidentFile.write(tempBuf, chunk) == chunk);
                }

                if (sendOk) {
                    ringTail = (ringTail + chunk) % RING_BUFFER_SIZE;
                    toDrain -= chunk;
                    failedAttempts = 0;
                    delay(4);
                } else {
                    failedAttempts++;
                    delay(5);
                }
            }

            if (hasActiveStream && !isUploadingArchive) {
                ws.sendTXT("STREAM_END");
                Serial.println("\n[SYS] Wi-Fi Stream completed (silence or max duration).");
                hasActiveStream = false;
            } else if (bleClientConnected) {
                sendBleText("STREAM_END");
                Serial.println("\n[SYS] BLE Stream completed to Mobile Companion.");
            }
            if (isOfflineRecording) {
                isOfflineRecording = false;
                if (offlineIncidentFile) {
                    offlineIncidentFile.close();
                    Serial.println("[FS] Offline recording saved and closed.");
                }
            }
            if (streamStartMs > 0) {
                totalStreamTimeMs += (millis() - streamStartMs);
                streamStartMs = 0;
            }
            isStreaming = false;
            WiFi.setSleep(false); // Maintain Wi-Fi modem awake in Standby: eliminates Windows WinError 121 and keeps connection permanently alive
            // Float tail back to pre-roll window
            ringTail = (ringHead + RING_BUFFER_SIZE - PRE_ROLL_BYTES) % RING_BUFFER_SIZE;
        } else {
            // Actively draining audio from ringTail to ringHead in steady, verified BATCH_SIZE chunks
            if (available >= BATCH_SIZE) {
                static unsigned long lastBatchSentAt = 0;
                unsigned long paceMs = (available > 16000) ? 12 : 35; // Catch up pre-roll smoothly without flooding network/BLE
                if (millis() - lastBatchSentAt >= paceMs) {
                    uint8_t txBuf[BATCH_SIZE];
                    for (size_t i = 0; i < BATCH_SIZE; i++) {
                        txBuf[i] = audioRingBuffer[(ringTail + i) % RING_BUFFER_SIZE];
                    }

                    bool sendOk = true;
                    if (hasActiveStream && ws.isConnected() && !isUploadingArchive) {
                        // Tier A
                        sendOk = ws.sendBIN(txBuf, BATCH_SIZE);
                    } else if (bleClientConnected && !isUploadingArchive) {
                        // Tier B
                        sendOk = sendBleChunk(txBuf, BATCH_SIZE);
                    } else if (isOfflineRecording && offlineIncidentFile) {
                        // Tier C
                        sendOk = (offlineIncidentFile.write(txBuf, BATCH_SIZE) == BATCH_SIZE);
                    }

                    if (sendOk) {
                        ringTail = (ringTail + BATCH_SIZE) % RING_BUFFER_SIZE;
                        lastBatchSentAt = millis();
                    } else {
                        delay(6); // Network/BLE congested: hold pointer and retry next tick
                    }
                }
            }
        }
    }

    // --- PIN KEEP-ALIVE & TELEMETRY HEARTBEAT (3.0 seconds) ---
    // Transmits regular keep-alive pings over Wi-Fi WebSocket or BLE Audio Bridge
    static unsigned long lastHeartbeatMs = 0;
    if (ws.isConnected()) {
        if (needImmediateHeartbeat || (millis() - lastHeartbeatMs >= 3000)) {
            needImmediateHeartbeat = false;
            lastHeartbeatMs = millis();
            int rssi = WiFi.RSSI();
            int batPct = calculateBatteryPct();
            int batMv = readBatteryMv();
            ws.sendTXT("PING:" + String(rssi) + ":" + String(batPct) + ":" + String(batMv));
        }
    } else if (bleClientConnected) {
        if (needImmediateHeartbeat || (millis() - lastHeartbeatMs >= 3000)) {
            needImmediateHeartbeat = false;
            lastHeartbeatMs = millis();
            int batPct = calculateBatteryPct();
            int batMv = readBatteryMv();
            sendBleText("PING:-55:" + String(batPct) + ":" + String(batMv));
        }
    }

    processArchiveUpload();

    // --- THERMAL & POWER MANAGEMENT ---
    // In dual-core mode, audio sampling runs on Core 1. When not actively streaming,
    // yield Core 0 to the FreeRTOS idle task (waiti) for 10ms to drop CPU load from 100% to ~2%
    // and keep the ESP32 cool to the touch!
    if (!isStreaming && !isUploadingArchive) {
        delay(10);
    }
}
